private enum class FileTab { ALL, VIDEO, IMAGE, MINE, DOCS }

@Composable
fun FilesTab(state: AppState, onUpload: () -> Unit) {
    val s = LocalStrings.current
    val home = state.home
    var tab by remember { mutableStateOf(FileTab.ALL) }
    var month by remember { mutableStateOf<String?>(null) }      // "2026-10" or null = all
    var day by remember { mutableStateOf<String?>(null) }        // "2026-10-14" or null
    var monthMenu by remember { mutableStateOf(false) }
    var searching by remember { mutableStateOf(false) }
    var query by remember { mutableStateOf("") }
    val clipboard = LocalClipboardManager.current
    val copied = tr("Link kopiert", "Link copied", "تم نسخ الرابط")

    val all = home?.files.orEmpty()
    val byTab = all.filter {
        when (tab) {
            FileTab.ALL -> !it.mine
            FileTab.VIDEO -> !it.mine && it.type == "Video"
            FileTab.IMAGE -> !it.mine && it.type == "Image"
            FileTab.MINE -> it.mine
            FileTab.DOCS -> !it.mine && (it.type == "Document" || it.type == "Link")
        }
    }
    val months = byTab.mapNotNull { it.createdAt?.take(7) }.distinct().sortedDescending()
    val list = byTab
        .filter {
            val d = fileDay(it.createdAt)
            when {
                day != null -> d == day
                month != null -> d?.startsWith(month!!) == true
                else -> true
            }
        }
        .filter { query.isBlank() || it.title().contains(query.trim(), ignoreCase = true) || (it.projectName ?: "").contains(query.trim(), ignoreCase = true) }

    LazyVerticalGrid(
        columns = GridCells.Fixed(3),
        contentPadding = bottomSpace,
        horizontalArrangement = Arrangement.spacedBy(8.dp),
        verticalArrangement = Arrangement.spacedBy(8.dp),
    ) {
        item(span = { GridItemSpan(3) }) {
            Row(Modifier.fillMaxWidth().padding(top = 6.dp), verticalAlignment = Alignment.CenterVertically) {
                Text(tr("Dateien", "Files", "الملفات"), color = C.Ink, style = T.Title1, modifier = Modifier.weight(1f))
                IconTile(if (searching) Ic.Close else Ic.Search, { searching = !searching; if (!searching) query = "" }, circle = true)
                Spacer(Modifier.width(8.dp))
                IconTile(Ic.Plus, onUpload, gold = true, circle = true, desc = tr("Fotos & Videos hochladen", "Upload photos & videos", "رفع صور وفيديو"))
            }
        }
        if (searching) item(span = { GridItemSpan(3) }) {
            Row(
                Modifier.fillMaxWidth().height(50.dp).clip(RoundedCornerShape(16.dp)).background(C.Surface.copy(alpha = 0.7f)).border(1.dp, C.Border, RoundedCornerShape(16.dp)).padding(horizontal = 14.dp),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                Icon(Ic.Search, null, tint = C.Muted, modifier = Modifier.size(19.dp))
                Spacer(Modifier.width(10.dp))
                Box(Modifier.weight(1f)) {
                    if (query.isEmpty()) Text(tr("Dateien durchsuchen", "Search files", "ابحث في الملفات"), color = C.Faint, style = T.Body)
                    BasicTextField(query, { query = it }, singleLine = true, textStyle = T.Body.copy(color = C.Ink), cursorBrush = SolidColor(C.Gold), modifier = Modifier.fillMaxWidth())
                }
            }
        }
        item(span = { GridItemSpan(3) }) { StorageMeter(state) }
        item(span = { GridItemSpan(3) }) {
            Row(Modifier.fillMaxWidth().padding(top = 4.dp)) {
                TypeTab(Ic.Grid, tab == FileTab.ALL, Modifier.weight(1f)) { tab = FileTab.ALL; month = null; day = null }
                TypeTab(Ic.Video, tab == FileTab.VIDEO, Modifier.weight(1f)) { tab = FileTab.VIDEO; month = null; day = null }
                TypeTab(Ic.Image, tab == FileTab.IMAGE, Modifier.weight(1f)) { tab = FileTab.IMAGE; month = null; day = null }
                TypeTab(Ic.FileUp, tab == FileTab.MINE, Modifier.weight(1f)) { tab = FileTab.MINE; month = null; day = null }
                TypeTab(Ic.File, tab == FileTab.DOCS, Modifier.weight(1f)) { tab = FileTab.DOCS; month = null; day = null }
            }
        }
        item(span = { GridItemSpan(3) }) {
            Row(Modifier.fillMaxWidth().padding(vertical = 6.dp), verticalAlignment = Alignment.CenterVertically) {
                Row(
                    Modifier.height(42.dp).clip(RoundedCornerShape(999.dp)).background(C.Surface.copy(alpha = 0.55f))
                        .border(1.dp, if (day != null || month != null) C.Gold.copy(alpha = 0.7f) else C.Border.copy(alpha = 0.55f), RoundedCornerShape(999.dp))
                        .clickable { monthMenu = true }.padding(horizontal = 14.dp),
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    Icon(Ic.Calendar, null, tint = C.Gold, modifier = Modifier.size(17.dp))
                    Spacer(Modifier.width(8.dp))
                    Text(
                        day?.let { fmtDate(it) } ?: month?.let { monthLabel(s.months, it) } ?: tr("Alle Daten", "All dates", "كل التواريخ"),
                        color = C.Ink, style = T.BodyS, fontWeight = FontWeight.SemiBold,
                    )
                    if (day != null || month != null) {
                        Spacer(Modifier.width(8.dp))
                        Icon(Ic.Close, null, tint = C.Muted, modifier = Modifier.size(16.dp).clickable { day = null; month = null })
                    } else {
                        Spacer(Modifier.width(6.dp))
                        Icon(Ic.ChevronDown, null, tint = C.Muted, modifier = Modifier.size(18.dp))
                    }
                }
                Spacer(Modifier.weight(1f))
                Row(Modifier.clip(RoundedCornerShape(10.dp)).clickable { clipboard.setText(AnnotatedString(Media.web("files"))); state.showToast(copied) }.padding(6.dp), verticalAlignment = Alignment.CenterVertically) {
                    Icon(Ic.Link, null, tint = C.Gold, modifier = Modifier.size(17.dp))
                    Spacer(Modifier.width(6.dp))
                    Text(tr("Abschnittslink kopieren", "Copy section link", "نسخ رابط القسم"), color = C.Gold, style = T.Footnote, fontWeight = FontWeight.SemiBold)
                }
            }
        }
        if (list.isEmpty()) {
            item(span = { GridItemSpan(3) }) {
                if (tab == FileTab.MINE) EmptyState(tr("Noch nichts hochgeladen", "Nothing uploaded yet", "لم ترفع شيئًا بعد"), tr("Über + kannst du Dateien oder ganze Ordner ans Team schicken.", "Use + to send files or whole folders to the team.", "يمكنك عبر زر + إرسال ملفات أو مجلدات كاملة إلى الفريق."), icon = UploadIcon)
                else EmptyState(s.noFiles, icon = Ic.Grid)
            }
        }
        items(list, key = { it.id }) { f ->
            MediaTile(f, Modifier.fillMaxWidth().aspectRatio(0.8f)) { state.go(Screen.Viewer(list.map { it.id }, list.indexOf(f))) }
        }
    }
    if (monthMenu) {
        FileDatePicker(
            days = byTab.mapNotNull { fileDay(it.createdAt) }.groupingBy { it }.eachCount(),
            selectedDay = day,
            onDay = { d -> day = d; month = null; monthMenu = false },
            onMonth = { m -> month = m; day = null; monthMenu = false },
            onAll = { day = null; month = null; monthMenu = false },
            onDismiss = { monthMenu = false },
        )
    }
}

fun monthLabel(months: List<String>, ym: String): String {
    val m = ym.substring(5, 7).toIntOrNull() ?: return ym
    return "${months[m - 1]} ${ym.take(4)}"
}

@Composable
private fun TypeTab(icon: ImageVector, selected: Boolean, modifier: Modifier, onClick: () -> Unit) {
    Column(modifier.clickable(onClick = onClick), horizontalAlignment = Alignment.CenterHorizontally) {
        Icon(icon, null, tint = if (selected) C.Gold else C.Muted, modifier = Modifier.padding(vertical = 10.dp).size(23.dp))
        Box(Modifier.fillMaxWidth().height(if (selected) 3.dp else 1.dp).background(if (selected) C.Gold else C.Border.copy(alpha = 0.6f)))
    }
}

/** Compact storage chip — slim glassy row with a thin gold progress bar (reels-app style). */
@Composable
fun StorageMeter(state: AppState) {
    val home = state.home
    val u = home?.usage
    val used = u?.storageUsedBytes ?: 0.0
    val limit = u?.storageLimitGb
    val ratio = if (limit != null && limit > 0) (used / (limit * 1e9)).toFloat().coerceIn(0.015f, 1f) else 0.015f
    Row(
        Modifier.fillMaxWidth().padding(vertical = 8.dp)
            .clip(RoundedCornerShape(999.dp))
            .background(C.Surface.copy(alpha = 0.55f))
            .border(1.dp, C.Border.copy(alpha = 0.55f), RoundedCornerShape(999.dp))
            .padding(start = 12.dp, end = 16.dp, top = 10.dp, bottom = 10.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Box(
            Modifier.size(32.dp).clip(CircleShape).background(Brush.linearGradient(listOf(Color(0xFFD9B44A), Color(0xFFF3DC8B)))),
            contentAlignment = Alignment.Center,
        ) { Icon(Ic.Zap, null, tint = C.Bg, modifier = Modifier.size(16.dp)) }
        Spacer(Modifier.width(10.dp))
        Column(Modifier.weight(1f)) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(gb(used), color = C.Ink, style = T.BodyS, fontWeight = FontWeight.SemiBold)
                if (limit != null) Text(" / ${num(limit)} GB", color = C.Muted, style = T.Caption)
                else {
                    Spacer(Modifier.width(6.dp))
                    Text(planName(state, home), color = C.Muted, style = T.Caption, maxLines = 1)
                }
            }
            Spacer(Modifier.height(5.dp))
            Box(Modifier.fillMaxWidth().height(5.dp).clip(RoundedCornerShape(999.dp)).background(C.Ink.copy(alpha = 0.14f))) {
                Box(Modifier.fillMaxWidth(ratio).height(5.dp).clip(RoundedCornerShape(999.dp)).background(Brush.horizontalGradient(listOf(Color(0xFFD9B44A), Color(0xFFF3DC8B)))))
            }
        }
        Spacer(Modifier.width(12.dp))
        u?.daysLeft?.let {
            Text(tr("Verlängerung in $it Tagen", "Renews in $it days", "تتجدد بعد ${arDays(it)}"), color = C.Gold, style = T.Micro, fontWeight = FontWeight.SemiBold)
        } ?: Text(
            tr("Kein aktives Paket", "No active plan", "لا توجد باقة نشطة"),
            color = C.Muted, style = T.Micro, fontWeight = FontWeight.SemiBold,
            textAlign = androidx.compose.ui.text.style.TextAlign.End,
        )
    }
}


/** The day a file arrived, in the phone's time zone ("YYYY-MM-DD"). */
fun fileDay(createdAt: String?): String? = epochOf(createdAt)?.let { de.focusontop.client.platform.localDateTime(it).take(10) }

/** Calendar to pick a day (or a whole month) — days with files carry a gold dot and their count. */
@Composable
private fun FileDatePicker(
    days: Map<String, Int>,
    selectedDay: String?,
    onDay: (String) -> Unit,
    onMonth: (String) -> Unit,
    onAll: () -> Unit,
    onDismiss: () -> Unit,
) {
    val s = LocalStrings.current
    val today = de.focusontop.client.platform.todayIso()
    val start = selectedDay?.take(7) ?: days.keys.maxOrNull()?.take(7) ?: today.take(7)
    var ym by remember { mutableStateOf(start) }
    val y = ym.take(4).toInt(); val m = ym.substring(5, 7).toInt()
    val lead = weekdayIndex(iso(y, m, 1))
    val count = daysInMonth(y, m)
    val inMonth = days.filterKeys { it.startsWith(ym) }.values.sum()
    androidx.compose.ui.window.Dialog(onDismissRequest = onDismiss) {
        Column(
            Modifier.fillMaxWidth().clip(RoundedCornerShape(26.dp)).background(C.Surface).border(1.dp, C.Border, RoundedCornerShape(26.dp)).padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(10.dp),
        ) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                IconTile(Ic.ChevronLeft, { ym = if (m == 1) "${y - 1}-12" else "$y-${(m - 1).toString().padStart(2, '0')}" }, size = 38.dp)
                Text("${s.months[m - 1]} $y", color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, textAlign = androidx.compose.ui.text.style.TextAlign.Center, modifier = Modifier.weight(1f))
                IconTile(Ic.ChevronRight, { ym = if (m == 12) "${y + 1}-01" else "$y-${(m + 1).toString().padStart(2, '0')}" }, size = 38.dp)
            }
            Row(Modifier.fillMaxWidth()) {
                shortWeekdays().forEach { Text(it, color = C.Muted, style = T.Caption, textAlign = androidx.compose.ui.text.style.TextAlign.Center, modifier = Modifier.weight(1f)) }
            }
            val cells = 7 * ((lead + count + 6) / 7)
            for (row in 0 until cells / 7) {
                Row(Modifier.fillMaxWidth()) {
                    for (col in 0 until 7) {
                        val d = row * 7 + col - lead + 1
                        Box(Modifier.weight(1f).height(46.dp), contentAlignment = Alignment.Center) {
                            if (d in 1..count) {
                                val date = iso(y, m, d)
                                val n = days[date] ?: 0
                                val sel = date == selectedDay
                                Column(
                                    Modifier.size(40.dp).clip(RoundedCornerShape(12.dp))
                                        .then(if (sel) Modifier.background(GoldBrush) else if (date == today) Modifier.border(1.dp, C.Gold.copy(alpha = 0.6f), RoundedCornerShape(12.dp)) else Modifier)
                                        .clickable(enabled = n > 0) { onDay(date) },
                                    horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.Center,
                                ) {
                                    Text(d.toString(), color = when { sel -> C.Bg; n > 0 -> C.Ink; else -> C.Faint }, style = T.Body, fontWeight = if (n > 0) FontWeight.SemiBold else FontWeight.Normal)
                                    if (n > 0 && !sel) Box(Modifier.padding(top = 2.dp).size(5.dp).clip(androidx.compose.foundation.shape.CircleShape).background(C.Gold))
                                }
                            }
                        }
                    }
                }
            }
            Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                GhostButton(tr("Alle Daten", "All dates", "كل التواريخ"), onAll, Modifier.weight(1f), small = true)
                if (inMonth > 0) GoldButton(tr("Ganzer Monat ($inMonth)", "Whole month ($inMonth)", "الشهر كامل ($inMonth)"), { onMonth(ym) }, Modifier.weight(1f), small = true)
            }
        }
    }
}
