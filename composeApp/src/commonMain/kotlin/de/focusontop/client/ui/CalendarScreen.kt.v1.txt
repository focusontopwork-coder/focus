@Composable
fun shortWeekdays(): List<String> = listOf(
    tr("Mo", "Mon", "ن"), tr("Di", "Tue", "ث"), tr("Mi", "Wed", "ر"), tr("Do", "Thu", "خ"),
    tr("Fr", "Fri", "ج"), tr("Sa", "Sat", "س"), tr("So", "Sun", "ح"),
)

private enum class CalView { DAY, WEEK, MONTH, YEAR }

@Composable
fun CalendarTab(state: AppState) {
    val s = LocalStrings.current
    val today = todayIso()
    var view by remember { mutableStateOf(CalView.MONTH) }
    var focus by remember { mutableStateOf(today) }          // the day everything is centred on
    var picked by remember { mutableStateOf<String?>(null) }
    val all = state.home?.appointments.orEmpty()
    val byDay = all.groupBy { it.date }
    val y = focus.take(4).toInt(); val m = focus.substring(5, 7).toInt()
    val wd = shortWeekdays()

    fun shift(dir: Int) {
        focus = when (view) {
            CalView.DAY -> addDaysIso(focus, dir)
            CalView.WEEK -> addDaysIso(focus, 7 * dir)
            CalView.MONTH -> { val nm = m + dir; val ny = y + if (nm < 1) -1 else if (nm > 12) 1 else 0; iso(ny, ((nm - 1 + 12) % 12) + 1, 1) }
            CalView.YEAR -> iso(y + dir, m, 1)
        }
        picked = null
    }

    val title = when (view) {
        CalView.DAY -> dayHeader(s, focus, today)
        CalView.WEEK -> { val ws = weekStart(focus); val we = addDaysIso(ws, 6); "${ws.substring(8, 10).toInt()}. – ${we.substring(8, 10).toInt()}. ${s.months[we.substring(5, 7).toInt() - 1]}" }
        CalView.MONTH -> "${s.months[m - 1]} $y"
        CalView.YEAR -> "$y"
    }
    // what the list under the calendar shows
    val listed: List<Appointment> = when {
        picked != null -> all.filter { it.date == picked }
        view == CalView.DAY -> all.filter { it.date == focus }
        view == CalView.WEEK -> { val ws = weekStart(focus); val we = addDaysIso(ws, 6); all.filter { it.date in ws..we } }
        view == CalView.MONTH -> all.filter { it.date.startsWith(focus.take(7)) && it.date >= minOf(today, focus.take(7) + "-01") }
        else -> all.filter { it.date >= today }.take(20)
    }

    LazyColumn(contentPadding = bottomSpace, verticalArrangement = Arrangement.spacedBy(12.dp)) {
        item {
            Row(Modifier.fillMaxWidth().padding(top = 10.dp), verticalAlignment = Alignment.CenterVertically) {
                Text(tr("Kalender", "Calendar", "التقويم"), color = C.Ink, style = T.Title1, modifier = Modifier.weight(1f))
                GoldButton(tr("Termin", "Book", "موعد"), { state.go(Screen.AppointmentRequest) }, icon = Ic.Plus, small = true)
            }
        }
        // Tag / Woche / Monat / Jahr
        item {
            Row(
                Modifier.fillMaxWidth().height(44.dp).clip(RoundedCornerShape(14.dp)).background(C.Surface.copy(alpha = 0.7f))
                    .border(1.dp, C.Border.copy(alpha = 0.6f), RoundedCornerShape(14.dp)).padding(4.dp),
            ) {
                listOf(CalView.DAY to tr("Tag", "Day", "يوم"), CalView.WEEK to tr("Woche", "Week", "أسبوع"), CalView.MONTH to tr("Monat", "Month", "شهر"), CalView.YEAR to tr("Jahr", "Year", "سنة")).forEach { (v, label) ->
                    val sel = view == v
                    Box(
                        Modifier.weight(1f).fillMaxHeight().clip(RoundedCornerShape(11.dp))
                            .then(if (sel) Modifier.background(GoldBrush) else Modifier)
                            .clickable { view = v; picked = null },
                        contentAlignment = Alignment.Center,
                    ) { Text(label, color = if (sel) C.Bg else C.Muted, style = T.BodyS, fontWeight = if (sel) FontWeight.SemiBold else FontWeight.SemiBold) }
                }
            }
        }
        // title row with arrows + "today"
        item {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(title, color = C.Ink, style = T.Title3, fontWeight = FontWeight.SemiBold, modifier = Modifier.weight(1f), maxLines = 1, overflow = TextOverflow.Ellipsis)
                if (!(focus == today || (view == CalView.MONTH && focus.take(7) == today.take(7)) || (view == CalView.YEAR && focus.take(4) == today.take(4)))) {
                    Text(
                        tr("Heute", "Today", "اليوم"), color = C.Gold, style = T.Footnote, fontWeight = FontWeight.SemiBold,
                        modifier = Modifier.clip(RoundedCornerShape(10.dp)).border(1.dp, C.Gold.copy(alpha = 0.5f), RoundedCornerShape(10.dp)).clickable { focus = today; picked = null }.padding(horizontal = 10.dp, vertical = 6.dp),
                    )
                    Spacer(Modifier.width(8.dp))
                }
                IconTile(Ic.ChevronLeft, { shift(-1) }, size = 38.dp)
                Spacer(Modifier.width(6.dp))
                IconTile(Ic.ChevronRight, { shift(1) }, size = 38.dp)
            }
        }
        item {
            when (view) {
                CalView.DAY, CalView.WEEK -> WeekStrip(weekStart(focus), today, if (view == CalView.DAY) focus else picked, byDay, wd) { d ->
                    if (view == CalView.DAY) focus = d else picked = if (picked == d) null else d
                }
                CalView.MONTH -> MonthGrid(y, m, today, picked, byDay, wd) { d -> picked = if (picked == d) null else d }
                CalView.YEAR -> YearGrid(y, today, byDay, s.months) { mm -> focus = iso(y, mm, 1); view = CalView.MONTH }
            }
        }
        if (view != CalView.YEAR) item {
            Row(horizontalArrangement = Arrangement.spacedBy(16.dp), verticalAlignment = Alignment.CenterVertically) {
                Legend(C.GreenText, tr("Bestätigt", "Confirmed", "مؤكد"))
                Legend(C.Gold, tr("Ausstehend", "Pending", "بانتظار التأكيد"))
            }
        }
        item {
            SectionTitle(
                when {
                    picked != null -> dayHeader(s, picked!!, today)
                    view == CalView.DAY -> tr("Termine an diesem Tag", "This day", "مواعيد هذا اليوم")
                    view == CalView.WEEK -> tr("Diese Woche", "This week", "هذا الأسبوع")
                    view == CalView.MONTH -> tr("In diesem Monat", "This month", "هذا الشهر")
                    else -> tr("Demnächst", "Upcoming", "القادم")
                },
                Modifier.padding(top = 4.dp),
                trailing = tr("${listed.size} Termine", "${listed.size} appointments", "${listed.size} مواعيد"),
            )
        }
        if (listed.isEmpty()) item {
            Column(
                Modifier.fillMaxWidth().clip(RoundedCornerShape(20.dp)).background(C.Surface.copy(alpha = 0.5f)).padding(22.dp),
                horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                Icon(Ic.Calendar, null, tint = C.Gold, modifier = Modifier.size(30.dp))
                Text(tr("Keine Termine", "No appointments", "لا توجد مواعيد"), color = C.Ink, style = T.Body, fontWeight = FontWeight.SemiBold)
                Text(tr("Brauchst du einen Dreh oder ein Meeting? Frag einfach an.", "Need a shoot or a meeting? Just ask.", "تحتاج تصويرًا أو اجتماعًا؟ اطلب موعدًا."), color = C.Muted, style = T.Footnote, textAlign = TextAlign.Center)
                GhostButton(tr("Termin anfragen", "Request appointment", "طلب موعد"), { state.go(Screen.AppointmentRequest) }, small = true, icon = Ic.CalendarPlus)
            }
        }
        items(listed, key = { it.id }) { a -> AppointmentRow(a) }
    }
}

@Composable
private fun Legend(color: androidx.compose.ui.graphics.Color, text: String) {
    Row(verticalAlignment = Alignment.CenterVertically) {
        Box(Modifier.size(7.dp).clip(CircleShape).background(color))
        Spacer(Modifier.width(6.dp))
        Text(text, color = C.Muted, style = T.Caption)
    }
}

/** Small dots under a day: green = confirmed, gold = waiting for confirmation (max 3). */
@Composable
private fun DayDots(list: List<Appointment>, onGold: Boolean = false) {
    if (list.isEmpty()) { Spacer(Modifier.height(5.dp)); return }
    Row(horizontalArrangement = Arrangement.spacedBy(2.dp)) {
        list.take(3).forEach { a ->
            Box(Modifier.size(5.dp).clip(CircleShape).background(if (onGold) C.Bg else if (a.status == "requested") C.Gold else C.GreenText))
        }
    }
}

@Composable
private fun WeekStrip(start: String, today: String, selected: String?, byDay: Map<String, List<Appointment>>, wd: List<String>, onPick: (String) -> Unit) {
    Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
        for (k in 0 until 7) {
            val d = addDaysIso(start, k)
            val sel = d == selected
            val isToday = d == today
            Column(
                Modifier.weight(1f).height(78.dp).clip(RoundedCornerShape(18.dp))
                    .then(if (sel) Modifier.background(GoldBrush) else Modifier.background(C.Surface.copy(alpha = 0.7f)))
                    .border(1.dp, if (isToday && !sel) C.Gold else C.Border.copy(alpha = 0.5f), RoundedCornerShape(18.dp))
                    .clickable { onPick(d) },
                horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(4.dp, Alignment.CenterVertically),
            ) {
                Text(wd[k], color = if (sel) C.Bg.copy(alpha = 0.8f) else C.Muted, style = T.Micro, fontWeight = FontWeight.SemiBold)
                Text(d.substring(8, 10).toInt().toString(), color = if (sel) C.Bg else if (isToday) C.Gold else C.Ink, style = T.Title3, fontWeight = FontWeight.SemiBold)
                DayDots(byDay[d].orEmpty(), onGold = sel)
            }
        }
    }
}

@Composable
private fun MonthGrid(y: Int, m: Int, today: String, picked: String?, byDay: Map<String, List<Appointment>>, wd: List<String>, onPick: (String) -> Unit) {
    val lead = weekdayIndex(iso(y, m, 1))
    val days = daysInMonth(y, m)
    val cells = 7 * ((lead + days + 6) / 7)
    Column(
        Modifier.fillMaxWidth().clip(RoundedCornerShape(24.dp)).background(Brush.verticalGradient(listOf(C.Surface2.copy(alpha = 0.9f), C.Surface.copy(alpha = 0.7f))))
            .border(1.dp, C.Border.copy(alpha = 0.5f), RoundedCornerShape(24.dp)).padding(horizontal = 8.dp, vertical = 12.dp),
    ) {
        Row(Modifier.fillMaxWidth().padding(bottom = 6.dp)) {
            wd.forEachIndexed { i, w -> Text(w, color = if (i >= 5) C.Gold.copy(alpha = 0.8f) else C.Muted, style = T.Caption, fontWeight = FontWeight.SemiBold, textAlign = TextAlign.Center, modifier = Modifier.weight(1f)) }
        }
        for (row in 0 until cells / 7) {
            Row(Modifier.fillMaxWidth()) {
                for (col in 0 until 7) {
                    val d = row * 7 + col - lead + 1
                    Box(Modifier.weight(1f).height(50.dp), contentAlignment = Alignment.Center) {
                        if (d in 1..days) {
                            val date = iso(y, m, d)
                            val list = byDay[date].orEmpty()
                            val isToday = date == today
                            val sel = date == picked
                            Column(
                                Modifier.size(42.dp).clip(CircleShape)
                                    .then(when { sel -> Modifier.background(GoldBrush); isToday -> Modifier.border(1.5.dp, C.Gold, CircleShape); else -> Modifier })
                                    .clickable { onPick(date) },
                                horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(2.dp, Alignment.CenterVertically),
                            ) {
                                Text(
                                    d.toString(), style = T.Body,
                                    color = when { sel -> C.Bg; isToday -> C.Gold; date < today -> C.Faint; else -> C.Ink },
                                    fontWeight = if (sel || isToday || list.isNotEmpty()) FontWeight.SemiBold else FontWeight.Medium,
                                )
                                DayDots(list, onGold = sel)
                            }
                        }
                    }
                }
            }
        }
    }
}

@Composable
private fun YearGrid(y: Int, today: String, byDay: Map<String, List<Appointment>>, months: List<String>, onMonth: (Int) -> Unit) {
    Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
        for (r in 0 until 4) {
            Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                for (c in 0 until 3) {
                    val mm = r * 3 + c + 1
                    val ym = "$y-${mm.toString().padStart(2, '0')}"
                    val n = byDay.filterKeys { it.startsWith(ym) }.values.sumOf { it.size }
                    val current = today.take(7) == ym
                    Column(
                        Modifier.weight(1f).clip(RoundedCornerShape(16.dp)).background(C.Surface.copy(alpha = 0.7f))
                            .border(1.dp, if (current) C.Gold else C.Border.copy(alpha = 0.5f), RoundedCornerShape(16.dp))
                            .clickable { onMonth(mm) }.padding(8.dp),
                        verticalArrangement = Arrangement.spacedBy(4.dp),
                    ) {
                        Row(verticalAlignment = Alignment.CenterVertically) {
                            Text(months[mm - 1].take(3), color = if (current) C.Gold else C.Ink, style = T.Footnote, fontWeight = FontWeight.SemiBold, modifier = Modifier.weight(1f))
                            if (n > 0) Text(n.toString(), color = C.Bg, style = T.Micro, fontWeight = FontWeight.SemiBold, modifier = Modifier.clip(CircleShape).background(C.Gold).padding(horizontal = 6.dp, vertical = 1.dp))
                        }
                        MiniMonth(y, mm, today, byDay)
                    }
                }
            }
        }
    }
}

/** Tiny month for the year view: today in gold, days with appointments marked. */
@Composable
private fun MiniMonth(y: Int, m: Int, today: String, byDay: Map<String, List<Appointment>>) {
    val lead = weekdayIndex(iso(y, m, 1))
    val days = daysInMonth(y, m)
    val cells = 7 * ((lead + days + 6) / 7)
    Column {
        for (row in 0 until cells / 7) {
            Row(Modifier.fillMaxWidth()) {
                for (col in 0 until 7) {
                    val d = row * 7 + col - lead + 1
                    Box(Modifier.weight(1f).height(12.dp), contentAlignment = Alignment.Center) {
                        if (d in 1..days) {
                            val date = iso(y, m, d)
                            val has = byDay[date].orEmpty().isNotEmpty()
                            Text(
                                d.toString(), fontSize = 7.sp, lineHeight = 8.sp,
                                color = when { date == today -> C.Gold; has -> C.GreenText; else -> C.Muted.copy(alpha = 0.8f) },
                                fontWeight = if (date == today || has) FontWeight.SemiBold else FontWeight.Normal,
                            )
                        }
                    }
                }
            }
        }
    }
}

@Composable
fun AppointmentRow(a: Appointment) {
    val wd = shortWeekdays()
    val requested = a.status == "requested"
    Row(
        Modifier.fillMaxWidth().height(IntrinsicSize.Min).clip(RoundedCornerShape(20.dp)).background(C.Surface.copy(alpha = 0.85f))
            .border(1.dp, C.Border.copy(alpha = 0.6f), RoundedCornerShape(20.dp)).padding(14.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Column(Modifier.width(70.dp)) {
            Text(if (a.allDay || a.time == null) LocalStrings.current.allDay else a.time, color = C.Gold, fontSize = if (a.time == null) 14.sp else 22.sp, fontWeight = FontWeight.SemiBold)
            Text("${wd[weekdayIndex(a.date)]}. ${a.date.substring(8, 10).toInt()}.", color = C.Muted, style = T.Caption)
        }
        Box(Modifier.width(1.dp).fillMaxHeight().background(C.Border))
        Spacer(Modifier.width(14.dp))
        Column(Modifier.weight(1f), verticalArrangement = Arrangement.spacedBy(3.dp)) {
            Text(a.title?.ifBlank { null } ?: "—", color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, maxLines = 2, overflow = TextOverflow.Ellipsis)
            val where = a.location?.takeIf { it.isNotBlank() } ?: if (!a.videoLink.isNullOrBlank()) "Online" else null
            where?.let { w ->
                Row(verticalAlignment = Alignment.CenterVertically, modifier = Modifier.clickable(enabled = !a.location.isNullOrBlank()) {
                    openUrl("https://maps.google.com/?q=" + (a.location ?: "").encodeForUrl())
                }) {
                    if (!a.location.isNullOrBlank()) { Icon(Ic.Pin, null, tint = C.Muted, modifier = Modifier.size(14.dp)); Spacer(Modifier.width(3.dp)) }
                    Text(w, color = C.Muted, style = T.Footnote, maxLines = 1, overflow = TextOverflow.Ellipsis)
                }
            }
            Row(verticalAlignment = Alignment.CenterVertically) {
                Box(Modifier.size(6.dp).clip(CircleShape).background(if (requested) C.Gold else C.GreenText))
                Spacer(Modifier.width(6.dp))
                Text(
                    if (requested) tr("Bestätigung ausstehend", "Awaiting confirmation", "بانتظار التأكيد") else tr("Bestätigt", "Confirmed", "مؤكَّد"),
                    color = if (requested) C.Gold else C.GreenText, style = T.Caption, fontWeight = FontWeight.SemiBold,
                )
            }
        }
        a.videoLink?.takeIf { it.isNotBlank() }?.let { link -> IconTile(Ic.Link, { openUrl(link) }, size = 40.dp, desc = LocalStrings.current.joinMeeting) }
    }
}

private data class Kind(val key: String, val minutes: Int)

@Composable
fun AppointmentRequestScreen(state: AppState) {
    val s = LocalStrings.current
    val scope = rememberCoroutineScope()
    val kinds = listOf(
        Kind("meeting", 60) to tr("Review-Meeting", "Review meeting", "اجتماع مراجعة"),
        Kind("shoot", 120) to tr("Foto-/Videodreh", "Photo / video shoot", "جلسة تصوير"),
        Kind("call", 30) to tr("Kurzer Call", "Quick call", "مكالمة سريعة"),
        Kind("consult", 60) to tr("Beratung", "Consultation", "استشارة"),
    )
    var kind by remember { mutableStateOf(kinds.first().first) }
    val today = todayIso()
    val days = remember(today) {
        val out = mutableListOf<String>(); var d = today
        while (out.size < 12) { if (weekdayIndex(d) != 6) out += d; d = nextDayIso(d) }
        out
    }
    var day by remember { mutableStateOf(days.firstOrNull { weekdayIndex(it) < 5 && it > today } ?: days.first()) }
    var slots by remember { mutableStateOf<List<Slot>?>(null) }
    var time by remember { mutableStateOf<String?>(null) }
    var note by remember { mutableStateOf("") }
    var busy by remember { mutableStateOf(false) }
    var error by remember { mutableStateOf<String?>(null) }
    val sent = tr("Anfrage gesendet – das Team meldet sich", "Request sent – the team will confirm", "أُرسل الطلب وسيؤكده الفريق")

    LaunchedEffect(day, kind) {
        slots = null; time = null
        slots = runCatching { state.api.availability(day, kind.minutes).slots.filter { it.time.endsWith(":00") } }.getOrElse { emptyList() }
    }
    val wd = shortWeekdays()
    Column(Modifier.fillMaxSize().screenBg().statusBarsPadding().navigationBarsPadding().imePadding()) {
        ScreenHeader(tr("Termin anfragen", "Request appointment", "طلب موعد"), onBack = { state.back() }, close = true)
        Column(Modifier.weight(1f).verticalScroll(rememberScrollState()).padding(horizontal = 16.dp), verticalArrangement = Arrangement.spacedBy(12.dp)) {
            StepLabel(1, tr("Art des Termins", "Type of appointment", "نوع الموعد"))
            kinds.chunked(2).forEach { row ->
                Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                    row.forEach { (k, label) -> ChoiceChip(label, kind == k, { kind = k }, Modifier.weight(1f), height = 52.dp) }
                }
            }
            StepLabel(2, tr("Tag", "Day", "اليوم") + " · " + s.months[day.substring(5, 7).toInt() - 1] + " " + day.take(4))
            LazyRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                items(days) { d ->
                    val sat = weekdayIndex(d) == 5
                    val sel = d == day
                    Column(
                        Modifier.width(56.dp).height(72.dp).clip(RoundedCornerShape(16.dp))
                            .then(if (sel) Modifier.background(GoldBrush) else Modifier.background(C.Surface.copy(alpha = 0.7f)))
                            .border(1.dp, if (sel) C.Gold else C.Border, RoundedCornerShape(16.dp))
                            .clickable(enabled = !sat) { day = d },
                        horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.Center,
                    ) {
                        Text(wd[weekdayIndex(d)], color = if (sel) C.Bg else C.Muted.copy(alpha = if (sat) 0.5f else 1f), style = T.Micro, fontWeight = FontWeight.SemiBold)
                        Text(d.substring(8, 10).toInt().toString(), color = if (sel) C.Bg else if (sat) C.Faint else C.Gold, style = T.Title2, fontWeight = FontWeight.SemiBold)
                    }
                }
            }
            StepLabel(3, tr("Uhrzeit · Berliner Zeit", "Time · Berlin time", "الوقت · توقيت برلين"))
            val list = slots
            if (list == null) Box(Modifier.fillMaxWidth().height(60.dp), contentAlignment = Alignment.Center) { CircularProgressIndicator(Modifier.size(24.dp), color = C.Gold) }
            else if (list.none { it.free }) Text(tr("An diesem Tag ist leider nichts frei.", "Nothing free on this day.", "لا توجد أوقات متاحة في هذا اليوم."), color = C.Muted, style = T.BodyS)
            else list.chunked(3).forEach { row ->
                Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                    row.forEach { sl -> ChoiceChip(sl.time, time == sl.time, { time = sl.time }, Modifier.weight(1f), enabled = sl.free, strike = !sl.free, height = 50.dp) }
                    repeat(3 - row.size) { Spacer(Modifier.weight(1f)) }
                }
            }
            StepLabel(4, tr("Hinweis ans Team · optional", "Note to the team · optional", "ملاحظة للفريق · اختياري"))
            Box(
                Modifier.fillMaxWidth().height(110.dp).clip(RoundedCornerShape(18.dp)).background(C.Surface.copy(alpha = 0.7f))
                    .border(1.dp, C.Border, RoundedCornerShape(18.dp)).padding(14.dp),
            ) {
                if (note.isEmpty()) Text(tr("z. B. Wir möchten das neue Reel vor der Veröffentlichung prüfen", "e.g. We'd like to review the new reel before it goes live", "مثال: نريد مراجعة الريلز الجديد قبل النشر"), color = C.Faint, style = T.BodyS)
                BasicTextField(note, { note = it }, textStyle = T.Body.copy(color = C.Ink), cursorBrush = SolidColor(C.Gold), modifier = Modifier.fillMaxSize())
            }
            error?.let { Text(it, color = C.Red, style = T.Footnote) }
            Spacer(Modifier.height(8.dp))
        }
        Column(Modifier.padding(horizontal = 16.dp, vertical = 10.dp), verticalArrangement = Arrangement.spacedBy(10.dp)) {
            Row(verticalAlignment = Alignment.Top) {
                Icon(Ic.Clock, null, tint = C.Muted, modifier = Modifier.size(16.dp))
                Spacer(Modifier.width(8.dp))
                Text(
                    tr("Das Team bestätigt innerhalb von 2 Arbeitsstunden. Der Termin erscheint im Kalender als „Bestätigung ausstehend“.",
                        "The team confirms within 2 working hours. Until then it shows in your calendar as “awaiting confirmation”.",
                        "يؤكد الفريق الطلب خلال ساعتي عمل، وسيظهر في التقويم بحالة «بانتظار التأكيد»."),
                    color = C.Muted, style = T.Caption,
                )
            }
            GoldButton(tr("Anfrage senden", "Send request", "إرسال الطلب"), {
                val t = time ?: return@GoldButton
                busy = true; error = null
                scope.launch {
                    try {
                        state.api.requestAppointment(kind.key, day, t, note.trim())
                        state.showToast(sent)
                        state.refresh()
                        state.back()
                    } catch (e: ApiError) {
                        error = if (e.status == 0) s.networkError else s.loadError
                    } finally { busy = false }
                }
            }, Modifier.fillMaxWidth(), enabled = time != null, loading = busy)
        }
