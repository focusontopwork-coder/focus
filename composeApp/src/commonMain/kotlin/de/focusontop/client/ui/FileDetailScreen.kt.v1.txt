/** One piece of work: watch / view it, approve or ask for a change, talk about it. */
@Composable
fun FileDetailScreen(state: AppState, initial: ClientFile) {
    val s = LocalStrings.current
    val scope = rememberCoroutineScope()
    var file by remember(initial.id) { mutableStateOf(initial) }
    val comments = remember(initial.id) { mutableStateListOf<Comment>() }
    var draft by remember { mutableStateOf("") }
    var busy by remember { mutableStateOf(false) }
    var askRevision by remember { mutableStateOf(false) }
    var confirmApprove by remember { mutableStateOf(false) }
    var toast by remember { mutableStateOf<String?>(null) }

    suspend fun loadComments() {
        runCatching { state.api.comments(file.id) }.onSuccess { comments.clear(); comments.addAll(it) }
    }

    LaunchedEffect(initial.id) {
        runCatching { state.api.file(initial.id) }.onSuccess { file = it }
        loadComments()
    }

    Column(Modifier.fillMaxSize().statusBarsPadding().navigationBarsPadding().imePadding()) {
        TopBar(file.title(), onBack = { state.back() })

        LazyColumn(
            Modifier.weight(1f).fillMaxWidth(),
            contentPadding = androidx.compose.foundation.layout.PaddingValues(horizontal = 16.dp, vertical = 4.dp),
            verticalArrangement = Arrangement.spacedBy(14.dp),
        ) {
            // ── the media itself
            item { MediaView(file) }

            // ── title + status
            item {
                Column(verticalArrangement = Arrangement.spacedBy(6.dp)) {
                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Text(file.title(), color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, modifier = Modifier.weight(1f))
                        val st = fileStatus(s, file.status)
                        Chip(st.label, st.fg)
                    }
                    Text(
                        listOfNotNull(file.taskName ?: file.projectName, fmtDate(file.createdAt), file.size).joinToString(" · "),
                        color = C.Muted, style = T.Footnote,
                    )
                }
            }

            // ── answer (only while it is waiting for the client)
            if (file.status == "ready_for_review") {
                item {
                    Column(
                        Modifier.fillMaxWidth().clip(RoundedCornerShape(20.dp)).background(C.Gold).padding(14.dp),
                        verticalArrangement = Arrangement.spacedBy(10.dp),
                    ) {
                        Text(s.waitingForYou, color = C.Bg, style = T.Body, fontWeight = FontWeight.SemiBold)
                        Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                            AnswerButton(s.approve, Icons.Filled.Check, filled = true, modifier = Modifier.weight(1f), enabled = !busy) { confirmApprove = true }
                            AnswerButton(s.requestChange, Icons.Filled.Edit, filled = false, modifier = Modifier.weight(1f), enabled = !busy) { askRevision = true }
                        }
                    }
                }
            }

            toast?.let { msg ->
                item {
                    Row(
                        Modifier.fillMaxWidth().clip(RoundedCornerShape(14.dp)).background(C.Green.copy(alpha = 0.18f)).padding(12.dp),
                        verticalAlignment = Alignment.CenterVertically,
                    ) {
                        Icon(Icons.Filled.CheckCircle, null, tint = C.GreenText)
                        Spacer(Modifier.width(8.dp))
                        Text(msg, color = C.Ink, style = T.BodyS, fontWeight = FontWeight.SemiBold)
                    }
                }
            }

            // ── the client's own change request
            if (!file.revisionNotes.isNullOrBlank()) {
                item {
                    Panel {
                        Column(verticalArrangement = Arrangement.spacedBy(6.dp)) {
                            Text(s.yourNotes, color = C.Gold, style = T.Footnote, fontWeight = FontWeight.SemiBold)
                            Text(file.revisionNotes ?: "", color = C.Ink, style = T.BodyS)
                        }
                    }
                }
            }

            // ── open / download (never the RAW source: only the delivered file)
            item {
                Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                    if (file.type != "Link") {
                        SecondaryButton(s.download, onClick = { Media.download(file.url, file.fileName ?: file.name)?.let(::openUrl) }, modifier = Modifier.weight(1f), icon = Icons.Filled.Download)
                    }
                    if (file.type == "Document" || file.type == "Link") {
                        SecondaryButton(s.open, onClick = { Media.abs(file.url)?.let(::openUrl) }, modifier = Modifier.weight(1f), icon = Icons.Filled.OpenInNew)
                    }
                }
            }

            // ── comments
            item { Text(s.comments, color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, modifier = Modifier.padding(top = 6.dp)) }
            items(comments, key = { it.id }) { c -> CommentRow(c) }
        }

        // ── write a comment
        Row(
            Modifier.fillMaxWidth().padding(horizontal = 12.dp, vertical = 10.dp),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            FtField("", draft, { draft = it }, modifier = Modifier.weight(1f), placeholder = s.writeComment, singleLine = false)
            Spacer(Modifier.width(8.dp))
            Box(
                Modifier.size(52.dp).clip(RoundedCornerShape(16.dp))
                    .background(if (draft.isBlank() || busy) C.Gold.copy(alpha = 0.4f) else C.Gold)
                    .clickable(enabled = draft.isNotBlank() && !busy) {
                        val text = draft.trim()
                        busy = true
                        scope.launch {
                            try { state.api.addComment(file.id, text); draft = ""; loadComments() }
                            catch (_: ApiError) { }
                            finally { busy = false }
                        }
                    },
                contentAlignment = Alignment.Center,
            ) { Icon(Icons.AutoMirrored.Filled.Send, s.send, tint = C.Bg) }
        }
    }

    if (confirmApprove) {
        ConfirmDialog(s.confirmApprove, onDismiss = { confirmApprove = false }) {
            confirmApprove = false
            busy = true
            scope.launch {
                try {
                    state.api.approve(file.id)
                    file = file.copy(status = "approved")
                    state.updateFile(file.id, "approved")
                    toast = s.approvedThanks
                    state.refresh()
                } catch (_: ApiError) { } finally { busy = false }
            }
        }
    }
    if (askRevision) {
        RevisionDialog(onDismiss = { askRevision = false }) { notes ->
            askRevision = false
            busy = true
            scope.launch {
                try {
                    state.api.requestRevision(file.id, notes)
                    file = file.copy(status = "revision_requested", revisionNotes = notes)
                    state.updateFile(file.id, "revision_requested", notes)
                    toast = s.revisionSent
                    state.refresh()
                } catch (_: ApiError) { } finally { busy = false }
            }
        }
    }
}

@Composable
private fun MediaView(f: ClientFile) {
    val shape = RoundedCornerShape(20.dp)
    when (f.type) {
        "Video" -> {
            val src = Media.video(f)
            if (src != null) VideoPlayer(src, Modifier.fillMaxWidth().aspectRatio(16f / 9f).clip(shape).background(C.Bg))
        }
        "Image" -> AsyncImage(
            model = Media.image(f.url, 1600),
            contentDescription = f.title(),
            contentScale = ContentScale.Fit,
            modifier = Modifier.fillMaxWidth().clip(shape).background(C.Bg),
        )
        else -> Box(
            Modifier.fillMaxWidth().height(150.dp).clip(shape).background(C.Surface)
                .clickable { Media.abs(f.url)?.let(::openUrl) },
            contentAlignment = Alignment.Center,
        ) {
            Column(horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(8.dp)) {
                Icon(Icons.Filled.Description, null, tint = C.Gold, modifier = Modifier.size(44.dp))
                Text(f.fileName ?: f.title(), color = C.Muted, style = T.Footnote, maxLines = 1)
            }
        }
    }
}

@Composable
private fun CommentRow(c: Comment) {
    val s = LocalStrings.current
    val mine = c.userRole == "client"
    Column(
        Modifier.fillMaxWidth().clip(RoundedCornerShape(16.dp))
            .background(if (mine) C.Gold.copy(alpha = 0.12f) else C.Surface).padding(12.dp),
        verticalArrangement = Arrangement.spacedBy(4.dp),
    ) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            Box(Modifier.size(8.dp).clip(CircleShape).background(if (mine) C.Gold else C.Blue))
            Spacer(Modifier.width(6.dp))
            Text(if (mine) (c.userName ?: "—") else s.team, color = C.Muted, style = T.Caption, fontWeight = FontWeight.SemiBold, modifier = Modifier.weight(1f))
            Text(fmtDate(c.createdAt), color = C.Faint, style = T.Micro)
        }
        Text(c.text ?: "", color = C.Ink, style = T.BodyS)
    }
}
