/** Focus On Top colours (same as the website / the app design). */

/** The deep-blue backdrop every screen sits on. */
fun Modifier.screenBg(): Modifier = this.background(Brush.verticalGradient(0f to C.BgTop, 0.45f to C.Bg, 1f to C.Bg))

val GoldBrush get() = Brush.verticalGradient(listOf(C.Gold, C.GoldDeep))

/** Lama Sans — the same font as the website (static weights 400–700; heavier asks use 700). */
@Composable
fun lamaFamily(): FontFamily = FontFamily(
    Font(Res.font.lama_400, FontWeight.Normal),
    Font(Res.font.lama_500, FontWeight.Medium),
    Font(Res.font.lama_600, FontWeight.SemiBold),
    Font(Res.font.lama_700, FontWeight.Bold),
    Font(Res.font.lama_700, FontWeight.ExtraBold),
    Font(Res.font.lama_700, FontWeight.Black),
)

@Composable
fun FtTheme(content: @Composable () -> Unit) {
    val lama = lamaFamily()
    val base = Typography()
    val type = Typography(
        displayLarge = base.displayLarge.copy(fontFamily = lama),
        displayMedium = base.displayMedium.copy(fontFamily = lama),
        displaySmall = base.displaySmall.copy(fontFamily = lama),
        headlineLarge = base.headlineLarge.copy(fontFamily = lama),
        headlineMedium = base.headlineMedium.copy(fontFamily = lama),
        headlineSmall = base.headlineSmall.copy(fontFamily = lama),
        titleLarge = base.titleLarge.copy(fontFamily = lama),
        titleMedium = base.titleMedium.copy(fontFamily = lama),
        titleSmall = base.titleSmall.copy(fontFamily = lama),
        bodyLarge = base.bodyLarge.copy(fontFamily = lama),
        bodyMedium = base.bodyMedium.copy(fontFamily = lama),
        bodySmall = base.bodySmall.copy(fontFamily = lama),
        labelLarge = base.labelLarge.copy(fontFamily = lama),
        labelMedium = base.labelMedium.copy(fontFamily = lama),
        labelSmall = base.labelSmall.copy(fontFamily = lama),
    )
    MaterialTheme(
        colorScheme = darkColorScheme(
            primary = C.Gold, onPrimary = C.Bg,
            background = C.Bg, onBackground = C.Ink,
            surface = C.Surface, onSurface = C.Ink,
            surfaceVariant = C.Surface, onSurfaceVariant = C.Muted,
            outline = C.Border, error = C.Red,
        ),
        typography = type,
    ) {
        // every Text without its own style picks up Lama Sans too
        ProvideTextStyle(type.bodyLarge, content)
    }
}

@Composable
fun PrimaryButton(text: String, onClick: () -> Unit, modifier: Modifier = Modifier, enabled: Boolean = true, loading: Boolean = false, icon: ImageVector? = null) {
    Button(
        onClick = onClick,
        enabled = enabled && !loading,
        modifier = modifier.fillMaxWidth().height(56.dp),
        shape = RoundedCornerShape(16.dp),
        colors = ButtonDefaults.buttonColors(containerColor = C.Gold, contentColor = C.Bg, disabledContainerColor = C.Gold.copy(alpha = 0.5f), disabledContentColor = C.Bg),
    ) {
        if (loading) {
            CircularProgressIndicator(Modifier.size(20.dp), color = C.Bg, strokeWidth = 2.5.dp)
        } else {
            if (icon != null) { Icon(icon, null, Modifier.size(20.dp)); Spacer(Modifier.width(8.dp)) }
            Text(text, fontSize = 17.sp, fontWeight = FontWeight.ExtraBold)
        }
    }
}

@Composable
fun SecondaryButton(text: String, onClick: () -> Unit, modifier: Modifier = Modifier, color: Color = C.Ink, icon: ImageVector? = null) {
    OutlinedButton(
        onClick = onClick,
        modifier = modifier.height(52.dp),
        shape = RoundedCornerShape(16.dp),
        border = BorderStroke(1.5.dp, C.Border),
        colors = ButtonDefaults.outlinedButtonColors(contentColor = color),
    ) {
        if (icon != null) { Icon(icon, null, Modifier.size(18.dp)); Spacer(Modifier.width(8.dp)) }
        Text(text, fontSize = 15.sp, fontWeight = FontWeight.Bold)
    }
}

@Composable
fun FtField(
    label: String,
    value: String,
    onChange: (String) -> Unit,
    modifier: Modifier = Modifier,
    placeholder: String = "",
    keyboard: KeyboardType = KeyboardType.Text,
    password: Boolean = false,
    singleLine: Boolean = true,
    minLines: Int = 1,
) {
    var visible by remember { mutableStateOf(false) }
    Column(modifier, verticalArrangement = Arrangement.spacedBy(8.dp)) {
        if (label.isNotEmpty()) Text(label, color = C.Muted, fontSize = 13.sp, fontWeight = FontWeight.Bold)
        OutlinedTextField(
            value = value,
            onValueChange = onChange,
            modifier = Modifier.fillMaxWidth(),
            placeholder = { Text(placeholder, color = C.Faint) },
            singleLine = singleLine,
            minLines = minLines,
            shape = RoundedCornerShape(16.dp),
            keyboardOptions = KeyboardOptions(keyboardType = if (password) KeyboardType.Password else keyboard),
            visualTransformation = if (password && !visible) PasswordVisualTransformation() else VisualTransformation.None,
            trailingIcon = if (password) {
                {
                    IconButton(onClick = { visible = !visible }) {
                        Icon(if (visible) Icons.Filled.VisibilityOff else Icons.Filled.Visibility, LocalStrings.current.showPassword, tint = C.Muted)
                    }
                }
            } else null,
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor = C.Gold, unfocusedBorderColor = C.Border,
                focusedContainerColor = C.Surface, unfocusedContainerColor = C.Surface,
                focusedTextColor = C.Ink, unfocusedTextColor = C.Ink, cursorColor = C.Gold,
            ),
        )
    }
}

@Composable
fun TopBar(title: String, onBack: (() -> Unit)? = null, actions: @Composable () -> Unit = {}) {
    Row(
        Modifier.fillMaxWidth().padding(horizontal = 16.dp, vertical = 12.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        if (onBack != null) {
            Box(
                Modifier.size(44.dp).clip(RoundedCornerShape(14.dp)).background(C.Surface).clickable(onClick = onBack),
                contentAlignment = Alignment.Center,
            ) { Icon(Icons.AutoMirrored.Filled.ArrowBack, LocalStrings.current.back, tint = C.Ink) }
            Spacer(Modifier.width(12.dp))
        }
        Text(title, color = C.Ink, fontSize = 20.sp, fontWeight = FontWeight.ExtraBold, modifier = Modifier.weight(1f))
        actions()
    }
}

@Composable
fun Chip(text: String, fg: Color, bg: Color = fg.copy(alpha = 0.16f)) {
    Text(
        text, color = fg, fontSize = 11.sp, fontWeight = FontWeight.ExtraBold,
        modifier = Modifier.clip(RoundedCornerShape(999.dp)).background(bg).padding(horizontal = 10.dp, vertical = 4.dp),
    )
}

@Composable
fun Panel(modifier: Modifier = Modifier, onClick: (() -> Unit)? = null, content: @Composable () -> Unit) {
    Box(
        modifier.fillMaxWidth().clip(RoundedCornerShape(20.dp)).background(C.Surface.copy(alpha = 0.85f))
            .border(1.dp, C.Border.copy(alpha = 0.6f), RoundedCornerShape(20.dp))
            .then(if (onClick != null) Modifier.clickable(onClick = onClick) else Modifier)
            .padding(14.dp),
    ) { content() }
}

@Composable
fun SelectPill(text: String, selected: Boolean, onClick: () -> Unit) {
    Text(
        text,
        color = if (selected) C.Bg else C.Muted,
        fontSize = 13.sp,
        fontWeight = FontWeight.ExtraBold,
        modifier = Modifier.clip(RoundedCornerShape(12.dp))
            .background(if (selected) C.Gold else C.Surface)
            .border(1.dp, if (selected) C.Gold else C.Border, RoundedCornerShape(12.dp))
            .clickable(onClick = onClick)
            .padding(horizontal = 14.dp, vertical = 9.dp),
    )
}

@Composable
fun EmptyState(title: String, body: String? = null, icon: ImageVector? = null) {
    Column(Modifier.fillMaxWidth().padding(vertical = 40.dp, horizontal = 24.dp), horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(8.dp)) {
        if (icon != null) Box(Modifier.size(56.dp).clip(RoundedCornerShape(18.dp)).background(C.Surface), contentAlignment = Alignment.Center) { Icon(icon, null, tint = C.Gold) }
        Text(title, color = C.Ink, fontSize = 16.sp, fontWeight = FontWeight.ExtraBold)
        if (body != null) Text(body, color = C.Muted, fontSize = 13.sp, lineHeight = 20.sp, textAlign = androidx.compose.ui.text.style.TextAlign.Center)
    }
}

@Composable
fun ErrorBanner(text: String, onRetry: () -> Unit) {
    Row(
        Modifier.fillMaxWidth().clip(RoundedCornerShape(14.dp)).background(C.Red.copy(alpha = 0.14f)).padding(12.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Text(text, color = C.Ink, fontSize = 13.sp, modifier = Modifier.weight(1f))
        Text(LocalStrings.current.retry, color = C.Gold, fontSize = 13.sp, fontWeight = FontWeight.ExtraBold, modifier = Modifier.clickable(onClick = onRetry).padding(6.dp))
    }
}
