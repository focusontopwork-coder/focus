/** One small globe button → a menu with the three languages. */
@Composable
fun LanguageSwitch(state: AppState) {
    var open by remember { mutableStateOf(false) }
    Box {
        Row(
            Modifier.height(40.dp).clip(RoundedCornerShape(999.dp))
                .background(Color.White.copy(alpha = 0.08f))
                .border(1.dp, Color.White.copy(alpha = 0.14f), RoundedCornerShape(999.dp))
                .clickable { open = true }
                .padding(start = 12.dp, end = 10.dp),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Icon(Icons.Filled.Language, null, tint = C.Ink, modifier = Modifier.size(18.dp))
            Spacer(Modifier.width(8.dp))
            Text(state.lang.code.uppercase(), color = C.Ink, style = T.Footnote, fontWeight = FontWeight.SemiBold)
            Spacer(Modifier.width(4.dp))
            Icon(Icons.Filled.KeyboardArrowDown, null, tint = C.Muted, modifier = Modifier.size(18.dp))
        }
        DropdownMenu(
            expanded = open,
            onDismissRequest = { open = false },
            modifier = Modifier.background(C.Surface).padding(horizontal = 6.dp),
        ) {
            Lang.entries.forEach { l ->
                val on = state.lang == l
                Row(
                    Modifier.width(196.dp).clip(RoundedCornerShape(12.dp))
                        .background(if (on) C.Gold.copy(alpha = 0.14f) else Color.Transparent)
                        .clickable { state.setLanguage(l); open = false }
                        .padding(horizontal = 12.dp, vertical = 12.dp),
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    Text(
                        l.code.uppercase(), color = if (on) C.Gold else C.Faint, style = T.Micro, fontWeight = FontWeight.SemiBold,
                        modifier = Modifier.width(28.dp),
                    )
                    Text(l.label, color = C.Ink, style = T.Body, fontWeight = if (on) FontWeight.SemiBold else FontWeight.SemiBold, modifier = Modifier.weight(1f))
                    if (on) Icon(Icons.Filled.Check, null, tint = C.Gold, modifier = Modifier.size(18.dp))
                }
            }
        }
    }
}

private fun loginError(s: Strings, e: ApiError): String = when {
    e.status == 0 -> s.networkError
    e.status == 401 -> s.wrongLogin
    e.code == "client_only" -> s.staffNotAllowed
    e.code == "pending" -> s.accountPending
    e.status == 403 && (e.message ?: "").let { it.contains("مراجعة") || it.contains("pending", true) || it.contains("Prüfung", true) || it.contains("geprüft", true) } -> s.accountPending
    e.status == 403 -> s.accountBlocked
    else -> e.message ?: s.loadError
}

@Composable
fun LoginScreen(state: AppState) {
    val s = LocalStrings.current
    val scope = rememberCoroutineScope()
    var email by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }
    var code by remember { mutableStateOf("") }
    var twoFaUser by remember { mutableStateOf<String?>(null) }
    var busy by remember { mutableStateOf(false) }
    var error by remember { mutableStateOf<String?>(null) }
    var config by remember { mutableStateOf<ClientConfig?>(null) }
    LaunchedEffect(Unit) { if (googleSignInAvailable || appleSignInAvailable) config = runCatching { state.api.clientConfig() }.getOrNull() }

    fun google() {
        val cfg = config ?: return
        val clientId = if (platformName == "ios") cfg.googleIosClientId ?: return else cfg.googleWebClientId ?: return
        error = null
        busy = true
        scope.launch {
            try {
                val googleToken = googleIdToken(clientId)
                if (googleToken != null) {
                    val resp = state.api.loginWithGoogle(googleToken)
                    if (resp.token == null || !state.onLoggedIn(resp)) error = s.staffNotAllowed
                }
            } catch (e: ApiError) {
                error = if (e.code == "google" || e.code == "google_not_configured" || e.status == 400) s.googleFailed else loginError(s, e)
            } catch (e: Exception) {
                error = s.googleFailed
            } finally {
                busy = false
            }
        }
    }

    fun apple() {
        error = null
        busy = true
        scope.launch {
            try {
                val appleToken = appleSignInToken()
                if (appleToken != null) {
                    val resp = state.api.loginWithApple(appleToken)
                    if (resp.token == null || !state.onLoggedIn(resp)) error = s.staffNotAllowed
                }
            } catch (e: ApiError) {
                error = loginError(s, e)
            } catch (e: Exception) {
                error = s.loadError
            } finally {
                busy = false
            }
        }
    }

    fun submit() {
        error = null
        busy = true
        scope.launch {
            try {
                val resp = if (twoFaUser != null) state.api.verify2fa(twoFaUser!!, code) else state.api.login(email, password)
                if (resp.requires2FA == true && resp.userId != null) {
                    twoFaUser = resp.userId
                } else if (resp.token != null) {
                    if (!state.onLoggedIn(resp)) error = s.staffNotAllowed
                } else error = s.wrongLogin
            } catch (e: ApiError) {
                error = loginError(s, e)
            } finally {
                busy = false
            }
        }
    }

    Column(
        Modifier.fillMaxSize().statusBarsPadding().navigationBarsPadding().imePadding().verticalScroll(rememberScrollState()).padding(horizontal = 24.dp, vertical = 16.dp),
        verticalArrangement = Arrangement.spacedBy(14.dp),
    ) {
        Row(Modifier.fillMaxWidth(), verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.SpaceBetween) {
            Image(painterResource(Res.drawable.app_icon), "Focus On Top", Modifier.size(44.dp).clip(CircleShape))
            LanguageSwitch(state)
        }
        Column(verticalArrangement = Arrangement.spacedBy(5.dp), modifier = Modifier.padding(top = 2.dp)) {
            Text(if (twoFaUser != null) s.twoFaTitle else s.hello, color = C.Ink, style = T.Title1)
            Text(if (twoFaUser != null) s.twoFaHint else s.loginHint, color = C.Muted, style = T.Body)
        }

        if (twoFaUser == null) {
            FtField(s.email, email, { email = it }, placeholder = "name@firma.de", keyboard = KeyboardType.Email)
            FtField(s.password, password, { password = it }, password = true)
            Text(s.forgotPassword, color = C.Gold, style = T.Footnote, fontWeight = FontWeight.SemiBold,
                modifier = Modifier.clickable { state.go(Screen.Forgot) })
        } else {
            FtField("", code, { code = it.filter(Char::isDigit).take(6) }, placeholder = "123456", keyboard = KeyboardType.NumberPassword)
        }

        if (error != null) {
            Text(error!!, color = C.Red, style = T.BodyS, fontWeight = FontWeight.SemiBold)
        }

        PrimaryButton(
            text = if (twoFaUser != null) s.verify else s.login,
            onClick = ::submit,
            loading = busy,
            enabled = if (twoFaUser != null) code.length == 6 else email.isNotBlank() && password.isNotBlank(),
        )

        if (twoFaUser == null) {
            Row(Modifier.fillMaxWidth(), verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.Center) {
                Text(tr("Noch kein Konto?", "No account yet?", "ليس لديك حساب؟"), color = C.Muted, style = T.Footnote)
                Spacer(Modifier.width(6.dp))
                Text(
                    tr("Konto erstellen", "Create account", "إنشاء حساب"),
                    color = C.Gold, style = T.Footnote, fontWeight = FontWeight.SemiBold,
                    modifier = Modifier.clickable { state.go(Screen.Register) },
                )
            }
        }

        if (twoFaUser == null && config?.googleSignIn == true) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Box(Modifier.weight(1f).height(1.dp).background(C.Border))
                Text(s.orDivider, color = C.Faint, style = T.Footnote, modifier = Modifier.padding(horizontal = 12.dp))
                Box(Modifier.weight(1f).height(1.dp).background(C.Border))
            }
            GoogleButton(s.continueWithGoogle, enabled = !busy, onClick = ::google)
        }

        if (twoFaUser == null && appleSignInAvailable && config?.googleSignIn == true) {
            Spacer(Modifier.height(10.dp))
            AppleButton(s.continueWithApple, enabled = !busy, onClick = ::apple)
        }

        Spacer(Modifier.height(12.dp))
        Column(Modifier.fillMaxWidth(), horizontalAlignment = Alignment.CenterHorizontally) {
            Text(s.clientsOnly, color = C.Faint, style = T.Caption, textAlign = TextAlign.Center)
            Text(s.privacy, color = C.Gold, style = T.Caption, fontWeight = FontWeight.SemiBold,
                modifier = Modifier.clickable { openUrl(Media.PRIVACY) }.padding(6.dp))
        }
    }
}

@Composable
fun ForgotScreen(state: AppState) {
    val s = LocalStrings.current
    val scope = rememberCoroutineScope()
    var email by remember { mutableStateOf("") }
    var busy by remember { mutableStateOf(false) }
    var done by remember { mutableStateOf(false) }
    var error by remember { mutableStateOf<String?>(null) }

    Column(Modifier.fillMaxSize().statusBarsPadding().navigationBarsPadding().imePadding()) {
        TopBar(s.resetTitle, onBack = { state.back() })
        Column(Modifier.padding(horizontal = 24.dp, vertical = 8.dp), verticalArrangement = Arrangement.spacedBy(20.dp)) {
            Text(s.resetHint, color = C.Muted, style = T.Body)
            if (done) {
                Panel { Text(s.resetSent, color = C.GreenText, style = T.BodyS, fontWeight = FontWeight.SemiBold) }
                SecondaryButton(s.back, onClick = { state.back() }, modifier = Modifier.fillMaxWidth())
            } else {
                FtField(s.email, email, { email = it }, placeholder = "name@firma.de", keyboard = KeyboardType.Email)
                if (error != null) Text(error!!, color = C.Red, style = T.BodyS)
                PrimaryButton(s.sendLink, loading = busy, enabled = email.contains("@"), onClick = {
                    busy = true; error = null
                    scope.launch {
                        try { state.api.forgotPassword(email); done = true }
                        catch (e: ApiError) { error = if (e.status == 0) s.networkError else s.loadError }
                        finally { busy = false }
                    }
                })
            }
        }
    }
}

@Composable

    var name by remember { mutableStateOf("") }
    var email by remember { mutableStateOf("") }
    var company by remember { mutableStateOf("") }

    var busy by remember { mutableStateOf(false) }
    var error by remember { mutableStateOf<String?>(null) }
    var done by remember { mutableStateOf(false) }
    var step by remember { mutableStateOf(0) }   // 0 = form, 1 = privacy choice, 2 = done
    var privateAccount by remember { mutableStateOf(true) }
    val errExists = tr("Diese E-Mail ist bereits registriert.", "This e-mail is already registered.", "هذا البريد مسجل بالفعل.")
    val errNet = tr("Keine Internetverbindung.", "No internet connection.", "لا يوجد اتصال بالإنترنت.")
    val errFail = tr("Registrierung fehlgeschlagen.", "Registration failed.", "تعذر إنشاء الحساب.")
    val backLabel = tr("Zurück zur Anmeldung", "Back to sign in", "العودة لتسجيل الدخول")
    val privacyTitle = tr("Kontotyp wählen", "Choose your account type", "اختر نوع حسابك")
    val privacySub = tr("Du kannst das später jederzeit in den Einstellungen ändern.", "You can change this anytime later in the settings.", "يمكنك تغيير ذلك لاحقاً من الإعدادات.")
    val pubTitle = tr("Öffentliches Konto", "Public account", "حساب عام")

    fun submit() {
        error = null
        busy = true
        scope.launch {
            try {
                state.api.register(name, email, password, company, privateAccount)
                done = true; step = 2
            } catch (e: ApiError) {
                error = when {
                    e.status == 400 && (e.message ?: "").contains("exists", true) ->

                }
            } catch (e: Exception) {

            } finally {
                busy = false
            }
        }
    }

    Column(

                    color = C.GreenText, style = T.BodyS, fontWeight = FontWeight.SemiBold,
                )
            }
            PrimaryButton(backLabel, onClick = { state.back() })
        } else if (step == 1) {
            Text(privacyTitle, color = C.Ink, style = T.Title2, fontWeight = FontWeight.SemiBold)
            Text(privacySub, color = C.Muted, style = T.Footnote)
            PrivacyOption(Ic.Globe, pubTitle, pubBody, selected = !privateAccount) { privateAccount = false }
            PrivacyOption(Ic.Lock, privTitle, privBody, selected = privateAccount) { privateAccount = true }
            if (error != null) Text(error!!, color = C.Red, style = T.BodyS, fontWeight = FontWeight.SemiBold)
            PrimaryButton(
                tr("Konto erstellen", "Create account", "إنشاء الحساب"),
                onClick = ::submit,
                loading = busy,
            )
            Text(
                tr("Zurück", "Back", "رجوع"),
                color = C.Muted, style = T.Footnote, fontWeight = FontWeight.SemiBold,
                modifier = Modifier.clickable { step = 0 }.padding(vertical = 6.dp),
            )
        } else {
            FtField(tr("Name", "Name", "الاسم"), name, { name = it })
            FtField(tr("E-Mail", "E-mail", "البريد الإلكتروني"), email, { email = it }, placeholder = "name@firma.de", keyboard = KeyboardType.Email)
            FtField(tr("Firma (optional)", "Company (optional)", "الشركة (اختياري)"), company, { company = it })
            FtField(tr("Passwort", "Password", "كلمة المرور"), password, { password = it }, password = true)
            if (error != null) Text(error!!, color = C.Red, style = T.BodyS, fontWeight = FontWeight.SemiBold)
            PrimaryButton(
                tr("Konto erstellen", "Create account", "إنشاء الحساب"),
                onClick = { error = null; step = 1 },
                loading = busy,
                enabled = name.isNotBlank() && email.contains("@") && password.length >= 6,
            )
            Text(
                tr("Bereits ein Konto? Anmelden", "Already have an account? Sign in", "لديك حساب؟ تسجيل الدخول"),
                color = C.Gold, style = T.Footnote, fontWeight = FontWeight.SemiBold,
                modifier = Modifier.clickable { state.back() }.padding(vertical = 6.dp),
            )
        }
    }
}


@Composable
private fun PrivacyOption(
    icon: androidx.compose.ui.graphics.vector.ImageVector,
    title: String, body: String, selected: Boolean, onClick: () -> Unit,
) {
    Row(
        Modifier.fillMaxWidth().clip(RoundedCornerShape(20.dp))
            .background(if (selected) C.Gold.copy(alpha = 0.10f) else C.Surface.copy(alpha = 0.6f))
            .border(if (selected) 1.5.dp else 1.dp, if (selected) C.Gold else C.Border, RoundedCornerShape(20.dp))
            .clickable(onClick = onClick).padding(14.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Box(
            Modifier.size(44.dp).clip(CircleShape).then(if (selected) Modifier.background(GoldBrush) else Modifier.background(C.Ink.copy(alpha = 0.07f))),
            contentAlignment = Alignment.Center,
        ) { Icon(icon, null, tint = if (selected) C.Bg else C.Gold, modifier = Modifier.size(21.dp)) }
        Spacer(Modifier.width(12.dp))
        Column(Modifier.weight(1f), verticalArrangement = Arrangement.spacedBy(3.dp)) {
            Text(title, color = C.Ink, style = T.Body, fontWeight = FontWeight.SemiBold)
            Text(body, color = C.Muted, style = T.Caption)
        }
        if (selected) Icon(Ic.CheckCircle, null, tint = C.Gold, modifier = Modifier.size(20.dp))
    }
}

@Composable
private fun GoogleButton(text: String, enabled: Boolean, onClick: () -> Unit) {
    Row(
        Modifier.fillMaxWidth().height(56.dp).clip(RoundedCornerShape(16.dp))
            .background(if (enabled) Color.White else Color.White.copy(alpha = 0.6f))
            .clickable(enabled = enabled, onClick = onClick),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.Center,
    ) {
        Text("G", color = Color(0xFF4285F4), style = T.Title2, fontWeight = FontWeight.SemiBold)
        Spacer(Modifier.width(12.dp))
        Text(text, color = Color(0xFF1F1F1F), style = T.Headline, fontWeight = FontWeight.SemiBold)
    }
}
