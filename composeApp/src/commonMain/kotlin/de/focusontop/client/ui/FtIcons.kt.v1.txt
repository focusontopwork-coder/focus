/**
 * "FT Cut" — Focus On Top's own icon language, drawn from scratch for this app.
 *
 *  • The cut: every container has its top-right corner sliced off at 45°,
 *    the same sharp angle as the FT logo.
 *  • The focus point: one small solid square per icon marks "the thing that
 *    matters" (the door, the date, the new message …) — a camera's focus mark.
 *  • Film language: files are a contact sheet, the calendar is a clapperboard,
 *    comments carry film perforations, sharing is an angular arrow with speed lines.
 *
 * Three layers, one colour: a faint body, a crisp outline, a solid accent —
 * Icon(tint) keeps each layer's own opacity.
 */
object Ic {
    private const val BODY = 0.2f
    private const val W = 1.75f

    private fun ft(
        name: String,
        body: List<String> = emptyList(),
        line: List<String> = emptyList(),
        accent: List<String> = emptyList(),
        mirror: Boolean = false,
        width: Float = W,
    ): ImageVector = ImageVector.Builder(name = name, defaultWidth = 24.dp, defaultHeight = 24.dp, viewportWidth = 24f, viewportHeight = 24f, autoMirror = mirror).apply {
        body.forEach { addPath(pathData = addPathNodes(it), fill = SolidColor(Color.Black), fillAlpha = BODY) }
        line.forEach {
            addPath(pathData = addPathNodes(it), fill = null, stroke = SolidColor(Color.Black),
                strokeLineWidth = width, strokeLineCap = StrokeCap.Round, strokeLineJoin = StrokeJoin.Round)
        }
        accent.forEach { addPath(pathData = addPathNodes(it), pathFillType = PathFillType.EvenOdd, fill = SolidColor(Color.Black)) }
    }.build()

    /** Fully filled (active tab) — the outline shape filled, with the accent cut out. */
    private fun filled(name: String, shape: String, holes: String = "", extra: List<String> = emptyList()) =
        ImageVector.Builder(name = name, defaultWidth = 24.dp, defaultHeight = 24.dp, viewportWidth = 24f, viewportHeight = 24f).apply {
            addPath(pathData = addPathNodes("$shape $holes"), pathFillType = PathFillType.EvenOdd, fill = SolidColor(Color.Black),
                stroke = SolidColor(Color.Black), strokeLineWidth = 1.4f, strokeLineJoin = StrokeJoin.Round)
            extra.forEach {
                addPath(pathData = addPathNodes(it), fill = null, stroke = SolidColor(Color.Black),
                    strokeLineWidth = W, strokeLineCap = StrokeCap.Round, strokeLineJoin = StrokeJoin.Round)
            }
        }.build()

    /** Small solid square — the "focus point". */
    private fun dot(x: Float, y: Float, s: Float = 2.4f) = "M${x - s / 2} ${y - s / 2}h${s}v${s}h${-s}z"
