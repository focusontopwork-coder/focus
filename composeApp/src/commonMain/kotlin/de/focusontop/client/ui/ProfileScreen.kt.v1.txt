private enum class ProfileGrid { POSTS, ALL, MINE }

@Composable
fun ProfileTab(state: AppState) {
    val home = state.home
    val user = home?.user ?: state.user
    val clipboard = LocalClipboardManager.current
    val copied = tr("Link kopiert", "Link copied", "تم نسخ الرابط")
    var grid by remember { mutableStateOf(ProfileGrid.POSTS) }
    val files = home?.files.orEmpty().filter {
        when (grid) {
            ProfileGrid.POSTS -> false
            ProfileGrid.ALL -> !it.mine
            ProfileGrid.MINE -> it.mine
        }
    }.take(60)
    val handle = (user?.company?.ifBlank { null } ?: user?.name ?: "").lowercase().replace(Regex("[^a-z0-9]+"), "_").trim('_')

    LazyVerticalGrid(
        columns = GridCells.Fixed(3),
        contentPadding = bottomSpace,
        horizontalArrangement = Arrangement.spacedBy(6.dp),
        verticalArrangement = Arrangement.spacedBy(6.dp),
    ) {
        item(span = { GridItemSpan(3) }) {
            Row(Modifier.fillMaxWidth().padding(top = 10.dp), verticalAlignment = Alignment.CenterVertically) {
                IconTile(Ic.Menu, { state.go(Screen.Settings) }, desc = tr("Einstellungen", "Settings", "الإعدادات"))
                Text(handle, color = C.Ink, style = T.Body, modifier = Modifier.weight(1f), textAlign = androidx.compose.ui.text.style.TextAlign.Center, maxLines = 1)
                IconTile(Ic.Link, { clipboard.setText(AnnotatedString(Media.web("profile"))); state.showToast(copied) })
            }
        }
        item(span = { GridItemSpan(3) }) {
            Row(Modifier.fillMaxWidth().padding(top = 12.dp), verticalAlignment = Alignment.CenterVertically) {
                InitialTile(user?.company?.ifBlank { null } ?: user?.name, 104.dp, user?.avatar, dashed = true)
                Spacer(Modifier.width(10.dp))
                val st = home?.stats
                val cm = state.me
                Stat(cm?.posts?.size ?: 0, tr("Beiträge", "Posts", "منشور"), Modifier.weight(1f))
                Stat(cm?.followers ?: 0, tr("Follower", "Followers", "متابِع"), Modifier.weight(1f))
                Stat(st?.files ?: 0, tr("Dateien", "Files", "ملف"), Modifier.weight(1f))
                Stat(home?.invoices?.size ?: 0, tr("Rechnungen", "Invoices", "فواتير"), Modifier.weight(1f).clickable { state.go(Screen.Invoices) })
            }
        }
        item(span = { GridItemSpan(3) }) {
            Column(Modifier.padding(top = 10.dp), verticalArrangement = Arrangement.spacedBy(3.dp)) {
                Text(user?.company?.ifBlank { null } ?: user?.name ?: "", color = C.Ink, style = T.Title2, fontWeight = FontWeight.SemiBold, maxLines = 1)
                val since = user?.createdAt?.take(4)
                val line = listOfNotNull(
                    since?.let { tr("Kunde seit $it", "Client since $it", "عميل لدى Focus On Top منذ $it") },
                    home?.manager?.name?.let { tr("Betreuer: $it", "Manager: $it", "مدير الحساب: $it") },
                ).joinToString(" · ")
                if (line.isNotEmpty()) Text(line, color = C.Muted, style = T.Footnote, maxLines = 2)
            }
        }
        item(span = { GridItemSpan(3) }) {
            Row(Modifier.padding(top = 8.dp), horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                GoldButton(tr("Profil bearbeiten", "Edit profile", "تعديل الحساب"), { state.go(Screen.Settings) }, Modifier.fillMaxWidth(), small = true)
            }
        }
        item(span = { GridItemSpan(3) }) { Box(Modifier.padding(top = 8.dp)) { PlanUsageCard(state, home) } }
        item(span = { GridItemSpan(3) }) {
            Column(Modifier.fillMaxWidth().padding(vertical = 8.dp), verticalArrangement = Arrangement.spacedBy(14.dp)) {
                Row(Modifier.fillMaxWidth()) {
                    Box(Modifier.weight(1f), contentAlignment = Alignment.Center) { CircleAction(Ic.Folder, tr("Projekte", "Projects", "المشاريع"), { state.go(Screen.Projects) }) }
                    Box(Modifier.weight(1f), contentAlignment = Alignment.Center) { CircleAction(Ic.Card, tr("Tarif", "Plan", "الباقة"), { state.go(Screen.Invoices) }) }
                    Box(Modifier.weight(1f), contentAlignment = Alignment.Center) { CircleAction(Ic.Calendar, tr("Termine", "Appointments", "المواعيد"), { state.go(Screen.AppointmentRequest) }) }
                }
            }
        }
        item(span = { GridItemSpan(3) }) {
            Row(Modifier.fillMaxWidth()) {
                GridTab(Ic.Image, grid == ProfileGrid.POSTS, Modifier.weight(1f)) { grid = ProfileGrid.POSTS }
                GridTab(Ic.Grid, grid == ProfileGrid.ALL, Modifier.weight(1f)) { grid = ProfileGrid.ALL }
                GridTab(Ic.FileUp, grid == ProfileGrid.MINE, Modifier.weight(1f)) { grid = ProfileGrid.MINE }
            }
        }
        if (grid == ProfileGrid.POSTS) {
            val posts = state.me?.posts.orEmpty()
            if (posts.isEmpty()) item(span = { GridItemSpan(3) }) {
                Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                    EmptyState(tr("Noch keine Beiträge", "No posts yet", "لا توجد منشورات بعد"))
                    GoldButton(tr("Beitrag posten", "Create a post", "انشر منشورًا"), { state.go(Screen.Compose("post")) }, Modifier.fillMaxWidth(), icon = Ic.Plus, small = true)
                }
            }
            items(posts, key = { "pp-" + it.id }) { p -> PostTile(p) { state.go(Screen.Post(p.id, p)) } }
        } else if (files.isEmpty()) item(span = { GridItemSpan(3) }) { EmptyState(LocalStrings.current.noFiles) }
        items(files, key = { "pf-" + it.id }) { f ->
            Box {
                MediaTile(f, Modifier.fillMaxWidth().aspectRatio(1f)) { state.go(Screen.Viewer(files.map { it.id }, files.indexOf(f))) }
                if (f.status == "ready_for_review") {
                    val cs = rememberCoroutineScope()
                    val s = LocalStrings.current
                    var busy by remember { mutableStateOf(false) }
                    Box(
                        Modifier.align(Alignment.BottomStart).padding(8.dp)
                            .clip(RoundedCornerShape(999.dp)).background(GoldBrush)
                            .clickable(enabled = !busy) {
                                busy = true
                                cs.launch {
                                    runCatching {
                                        state.api.approve(f.id)
                                        state.updateFile(f.id, "approved")
                                        state.showToast(s.approvedThanks)
                                        state.refresh()
                                    }
                                    busy = false
                                }
                            }.padding(horizontal = 12.dp, vertical = 7.dp),
                    ) {
                        Text(if (busy) "…" else tr("Freigeben", "Approve", "موافق"), color = C.Bg, style = T.Caption, fontWeight = FontWeight.SemiBold)
                    }
                }
            }
        }
    }
}

@Composable
private fun Stat(n: Int, label: String, modifier: Modifier) {
    Column(modifier, horizontalAlignment = Alignment.CenterHorizontally) {
        Text(n.toString(), color = C.Ink, style = T.Title2, fontWeight = FontWeight.SemiBold)
        Text(label, color = C.Muted, style = T.Caption)
    }
}

@Composable
private fun GridTab(icon: ImageVector, selected: Boolean, modifier: Modifier, onClick: () -> Unit) {
    Column(modifier.clickable(onClick = onClick), horizontalAlignment = Alignment.CenterHorizontally) {
        Icon(icon, null, tint = if (selected) C.Gold else C.Muted, modifier = Modifier.padding(vertical = 10.dp).size(22.dp))
        Box(Modifier.fillMaxWidth().height(if (selected) 3.dp else 1.dp).background(if (selected) C.Gold else C.Border.copy(alpha = 0.6f)))
    }
}

/** Plan with the three usage meters (storage, reels, shooting hours). */
@Composable
fun PlanUsageCard(state: AppState, home: Home?) {
    val sub = home?.subscription
    val u = home?.usage ?: Usage()
    Column(
        Modifier.fillMaxWidth().clip(RoundedCornerShape(22.dp)).background(C.Surface.copy(alpha = 0.85f))
            .border(1.dp, C.Border.copy(alpha = 0.6f), RoundedCornerShape(22.dp)).padding(14.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp),
    ) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            Box(Modifier.size(42.dp).clip(RoundedCornerShape(13.dp)).border(1.dp, C.Border, RoundedCornerShape(13.dp)), contentAlignment = Alignment.Center) {
                Icon(Ic.Shield, null, tint = C.Gold, modifier = Modifier.size(20.dp))
            }
            Spacer(Modifier.width(12.dp))
            Column(Modifier.weight(1f)) {
                Text(planName(state, home), color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, maxLines = 1)
                val line = listOfNotNull(
                    u.monthsTotal?.let { tr("$it-Monats-Abo", "$it-month plan", "اشتراك $it أشهر") },
                    u.monthIndex?.let { tr("Monat $it", "Month $it", "الشهر $it") },
                    sub?.endDate?.let { tr("verlängert ${fmtDate(it)}", "renews ${fmtDate(it)}", "يتجدد ${fmtDate(it)}") },
                ).joinToString(" · ")
                if (line.isNotEmpty()) Text(line, color = C.Muted, style = T.Caption, maxLines = 2)
            }
            if (sub?.status == "active") OutlinePill(tr("Aktiv", "Active", "نشطة"))
        }
        Row(horizontalArrangement = Arrangement.spacedBy(14.dp)) {
            UsageMeter(tr("Speicher", "Storage", "التخزين"), "${gb(u.storageUsedBytes)} GB", u.storageLimitGb?.let { "/ ${num(it)}" }, u.storageLimitGb?.let { (u.storageUsedBytes / (it * 1e9)).toFloat() }, Modifier.weight(1f))
            UsageMeter(tr("Reels/Monat", "Reels/month", "ريلز الشهر"), u.reelsThisMonth.toString(), u.reelsLimit?.let { "/ ${num(it)}" }, u.reelsLimit?.let { (u.reelsThisMonth / it).toFloat() }, Modifier.weight(1f))
            UsageMeter(tr("Drehstunden", "Shoot hours", "ساعات التصوير"), "${num(u.shootHoursThisMonth)} h", u.shootHoursLimit?.let { "/ ${num(it)} h" }, u.shootHoursLimit?.let { (u.shootHoursThisMonth / it).toFloat() }, Modifier.weight(1f))
        }
        Row(verticalAlignment = Alignment.CenterVertically) {
            Row(Modifier.weight(1f).clickable { openUrl(Media.web("subscriptions")) }, verticalAlignment = Alignment.CenterVertically) {
                Text(tr("Tarife vergleichen", "Compare plans", "مقارنة الباقات"), color = C.Gold, style = T.BodyS, fontWeight = FontWeight.SemiBold)
                Spacer(Modifier.width(4.dp))
                Icon(Ic.Forward, null, tint = C.Gold, modifier = Modifier.size(16.dp))
            }
            val draft = tr("Ich interessiere mich für ein Upgrade meines Tarifs.", "I'm interested in upgrading my plan.", "أرغب في ترقية باقتي.")
            GoldButton("Upgrade", { state.go(Screen.Chat(draft = draft)) }, small = true)
        }
    }
}

@Composable
private fun UsageMeter(label: String, value: String, limit: String?, fraction: Float?, modifier: Modifier) {
    Column(modifier, verticalArrangement = Arrangement.spacedBy(5.dp)) {
        Text(label, color = C.Muted, style = T.Micro, maxLines = 1)
        Row(verticalAlignment = Alignment.Bottom) {
            Text(value, color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, maxLines = 1)
            if (limit != null) Text(" $limit", color = C.Muted, style = T.Caption, maxLines = 1)
        }
        SegmentMeter(fraction ?: 0f, segments = 8, height = 4.dp)
    }
}

/* ───────────── settings ───────────── */

@Composable
fun SettingsScreen(state: AppState) {
    val s = LocalStrings.current
    val scope = rememberCoroutineScope()
    val home = state.home
    val user = home?.user ?: state.user
    var confirmLogout by remember { mutableStateOf(false) }
    var confirmDelete by remember { mutableStateOf(false) }
    var deleteDone by remember { mutableStateOf(false) }
    var langMenu by remember { mutableStateOf(false) }
    var notifOn by remember { mutableStateOf(Prefs.get("notif_off") != "1") }

    Column(Modifier.fillMaxSize().screenBg().statusBarsPadding().navigationBarsPadding()) {
        ScreenHeader(tr("Einstellungen", "Settings", "الإعدادات والنشاط"), onBack = { state.back() })
        LazyColumn(contentPadding = PaddingValues(start = 16.dp, end = 16.dp, bottom = 40.dp), verticalArrangement = Arrangement.spacedBy(8.dp)) {
            item {
                Row(
                    Modifier.fillMaxWidth().clip(RoundedCornerShape(22.dp)).background(C.Surface.copy(alpha = 0.85f)).border(1.dp, C.Border.copy(alpha = 0.6f), RoundedCornerShape(22.dp)).padding(14.dp),
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    InitialTile(user?.company?.ifBlank { null } ?: user?.name, 60.dp, user?.avatar, dashed = true)
                    Spacer(Modifier.width(12.dp))
                    Column(Modifier.weight(1f)) {
                        Text(user?.company?.ifBlank { null } ?: user?.name ?: "", color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, maxLines = 1)
                        val days = home?.usage?.daysLeft
                        Text(listOfNotNull(planName(state, home), days?.let { tr("verlängert in $it Tagen", "renews in $it days", "تتجدد بعد ${arDays(it)}") }).joinToString(" · "), color = C.Muted, style = T.Caption, maxLines = 2)
                    }
                    if (home?.subscription?.status == "active") OutlinePill(tr("Aktiv", "Active", "نشطة"))
                }
            }
            item { Label(tr("Dein Konto", "Your account", "حسابك")) }
            item {
                MenuGroup {
                    MenuRow(Ic.User, tr("Persönliche Daten", "Personal details", "المعلومات الشخصية")) { openUrl(Media.web("profile")) }
                    MenuRow(Ic.Lock, tr("Sicherheit & 2-Faktor", "Security & 2-factor", "الأمان والتحقق بخطوتين")) { openUrl(Media.web("profile")) }
                    MenuRow(Ic.Card, tr("Tarif, Rechnungen & Zahlung", "Plan, invoices & payment", "الباقة والفواتير وطريقة الدفع")) { state.go(Screen.Invoices) }
                }
            }
            item { Label(tr("Community", "Community", "المجتمع")) }
            item { PrivacyRow(state) }
            item { Label(tr("Einstellungen", "Preferences", "تفضيلاتك")) }
            item {
                MenuGroup {
                    MenuRow(Ic.Bell, tr("Mitteilungen", "Notifications", "الإشعارات"), trailing = {
                        Switch(notifOn, { on ->
                            notifOn = on; Prefs.set("notif_off", if (on) null else "1")
                            if (on) requestNotificationPermission()
                        }, colors = SwitchDefaults.colors(checkedThumbColor = C.Bg, checkedTrackColor = C.Gold, uncheckedTrackColor = C.Border))
                    }) { notifOn = !notifOn; Prefs.set("notif_off", if (notifOn) null else "1") }
                    Box {
                        MenuRow(Ic.Globe, s.language, value = state.lang.label) { langMenu = true }
                        DropdownMenu(langMenu, { langMenu = false }, containerColor = C.Surface) {
                            Lang.entries.forEach { l -> DropdownMenuItem(text = { Text(l.label, color = if (l == state.lang) C.Gold else C.Ink) }, onClick = { state.setLanguage(l); langMenu = false }) }
                        }
                    }
                    MenuRow(Ic.Moon, tr("Darstellung", "Appearance", "المظهر"), value = tr("Dunkel", "Dark", "داكن")) {}
                }
            }
            item { Label(tr("Tools nach Berechtigung", "Tools", "الأدوات حسب صلاحياتك")) }
            item {
                MenuGroup {
                    MenuRow(Ic.Sparkles, "Trendo AI") { openUrl(Media.web("trendo")) }
                    MenuRow(Ic.Tasks, tr("Projekte & Aufgaben", "Projects & tasks", "المشاريع والمهام")) { state.go(Screen.Projects) }
                    MenuRow(Ic.Receipt, tr("Rechnungen", "Invoices", "الفواتير")) { state.go(Screen.Invoices) }
                }
            }
            item { Label(tr("Support", "Support", "الدعم")) }
            item {
                MenuGroup {
                    MenuRow(Ic.Help, tr("Hilfe & Support kontaktieren", "Help & contact support", "المساعدة والتواصل مع الدعم")) { state.go(Screen.Chat()) }
                    MenuRow(Ic.Shield, s.privacy) { openUrl(Media.PRIVACY) }
                }
            }
            item { Spacer(Modifier.height(6.dp)) }
            item { GhostButton(s.logout, { confirmLogout = true }, Modifier.fillMaxWidth(), icon = Ic.Logout, color = C.Red) }
            item {
                Text(
                    if (deleteDone) s.deleteRequested else s.deleteAccount, color = C.Faint, style = T.Footnote,
                    modifier = Modifier.fillMaxWidth().clickable(enabled = !deleteDone) { confirmDelete = true }.padding(12.dp),
                    textAlign = androidx.compose.ui.text.style.TextAlign.Center,
                )
            }
        }
    }
    if (confirmLogout) ConfirmDialog(s.confirmLogout, onDismiss = { confirmLogout = false }) { confirmLogout = false; state.signOut() }
    if (confirmDelete) ConfirmDialog(s.deleteAccountBody, onDismiss = { confirmDelete = false }) {
        confirmDelete = false
        scope.launch { runCatching { state.api.requestDelete() }.onSuccess { deleteDone = true } }
    }
}

@Composable
private fun Label(text: String) {
    Text(text, color = C.Muted, style = T.Footnote, modifier = Modifier.padding(start = 4.dp, top = 10.dp))
}

/* ───────────── invoices ───────────── */

@Composable
fun InvoicesScreen(state: AppState) {
    val s = LocalStrings.current
    val home = state.home
    val list = home?.invoices.orEmpty()
    val today = todayIso()
    Column(Modifier.fillMaxSize().screenBg().statusBarsPadding().navigationBarsPadding()) {
        ScreenHeader(s.tabInvoices, onBack = { state.back() })
        LazyColumn(contentPadding = PaddingValues(start = 16.dp, end = 16.dp, bottom = 40.dp), verticalArrangement = Arrangement.spacedBy(10.dp)) {
            val sub = home?.subscription
            if (sub?.contractStatus == "sent" && !sub.contractToken.isNullOrBlank()) {
                item { GoldButton(s.signContract, { openUrl(Media.contract(sub.contractToken)) }, Modifier.fillMaxWidth()) }
            }
            if (list.isEmpty()) item { EmptyState(s.noInvoices, icon = Ic.Receipt) }
            items(list, key = { it.id }) { inv ->
                val st = invoiceStatus(s, inv, today)
                val pdf = Media.download(inv.attachmentUrl, "${s.invoice}-${inv.invoiceNumber ?: inv.id}.pdf")
                Row(
                    Modifier.fillMaxWidth().clip(RoundedCornerShape(20.dp)).background(C.Surface.copy(alpha = 0.85f)).border(1.dp, C.Border.copy(alpha = 0.6f), RoundedCornerShape(20.dp))
                        .clickable(enabled = pdf != null) { pdf?.let(::openUrl) }.padding(14.dp),
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    Box(Modifier.size(46.dp).clip(RoundedCornerShape(15.dp)).background(C.Bg.copy(alpha = 0.4f)), contentAlignment = Alignment.Center) { Icon(Ic.File, null, tint = C.Gold) }
                    Spacer(Modifier.width(12.dp))
                    Column(Modifier.weight(1f), verticalArrangement = Arrangement.spacedBy(4.dp)) {
                        Text("${s.invoice} ${inv.invoiceNumber ?: ""}".trim(), color = C.Ink, style = T.Body, fontWeight = FontWeight.SemiBold)
                        Row(verticalAlignment = Alignment.CenterVertically) {
                            Chip(st.label, st.fg)
                            Spacer(Modifier.width(8.dp))
                            Text("${s.due}: ${fmtDate(inv.dueDate)}", color = C.Muted, style = T.Caption)
                        }
                    }
                    Text(fmtMoney(invoiceTotal(inv), inv.currency), color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold)
                }
            }
        }
    }
}

/* ───────────── projects & tasks ───────────── */

@Composable
fun ProjectsScreen(state: AppState) {
    val s = LocalStrings.current
    val tasks = state.home?.tasks.orEmpty()
    val other = tr("Weitere", "Other", "أخرى")
    Column(Modifier.fillMaxSize().screenBg().statusBarsPadding().navigationBarsPadding()) {
        ScreenHeader(tr("Projekte & Aufgaben", "Projects & tasks", "المشاريع والمهام"), onBack = { state.back() })
        LazyColumn(contentPadding = PaddingValues(start = 16.dp, end = 16.dp, bottom = 40.dp), verticalArrangement = Arrangement.spacedBy(10.dp)) {
            if (tasks.isEmpty()) item { EmptyState(s.noTasks, icon = Ic.Tasks) }
            tasks.groupBy { it.projectName ?: other }.forEach { (project, list) ->
                item(key = "p-$project") { Text(project, color = C.Gold, style = T.Footnote, fontWeight = FontWeight.SemiBold, modifier = Modifier.padding(top = 8.dp)) }
                items(list, key = { it.id }) { t ->
                    val st = taskStatus(s, t.status)
                    Panel {
                        Column(verticalArrangement = Arrangement.spacedBy(6.dp)) {
                            Row(verticalAlignment = Alignment.CenterVertically) {
                                Text(t.title ?: "—", color = C.Ink, style = T.Body, fontWeight = FontWeight.SemiBold, modifier = Modifier.weight(1f), maxLines = 2, overflow = TextOverflow.Ellipsis)
                                Spacer(Modifier.width(8.dp))
                                Chip(st.label, st.fg)
                            }
                            if (!t.deliveryDate.isNullOrBlank()) Text("${s.delivery}: ${fmtDate(t.deliveryDate)}", color = C.Muted, style = T.Caption)
                        }
                    }
                }
            }
        }
    }
}
