/**
 * Reels browser — TikTok/Instagram-style: full-screen vertical video pager of
 * the delivered videos, snap-scrolling, one video playing at a time.
 * The files grid stays reachable through the grid toggle in the top bar.
 */
@Composable
fun ReelsScreen(state: AppState, onGrid: () -> Unit, onUpload: () -> Unit) {
    val videos = state.home?.files.orEmpty().filter { it.type == "Video" && !it.mine }

    if (videos.isEmpty()) {
        Column(Modifier.fillMaxSize()) {
            ReelsTopBar(onGrid)
            EmptyState(tr("Keine Videos", "No videos", "لا توجد فيديوهات"), icon = Ic.Video)
        }
        return
    }

    val pager = rememberLazyListState()
    val current by remember { derivedStateOf { pager.firstVisibleItemIndex } }

    Box(Modifier.fillMaxSize().background(Color.Black)) {
        BoxWithConstraints(Modifier.fillMaxSize()) {
            val pageH = maxHeight
            LazyColumn(
                state = pager,
                modifier = Modifier.fillMaxSize(),
                flingBehavior = rememberSnapFlingBehavior(lazyListState = pager),
            ) {
                items(videos.size) { i ->
                    val f = videos[i]
                    Box(Modifier.fillMaxWidth().height(pageH)) {
                        Media.video(f)?.let { src ->
                            ReelPlayer(src, playing = i == current, modifier = Modifier.fillMaxSize())
                        } ?: Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                            CircularProgressIndicator(color = C.Gold)
                        }
                        // تعتيم سفلي خفيف لقراءة النصوص
                        Box(
                            Modifier.fillMaxWidth().height(150.dp).align(Alignment.BottomCenter)
                                .background(Brush.verticalGradient(listOf(Color.Transparent, Color.Black.copy(alpha = 0.55f)))),
                        )
                        // معلومات الفيديو
                        Column(
                            Modifier.align(Alignment.BottomStart).padding(start = 16.dp, end = 78.dp, bottom = 20.dp),
                        ) {
                            Text(f.title(), color = Color.White, style = T.Headline, fontWeight = FontWeight.SemiBold, maxLines = 2)
                            fileDay(f.createdAt)?.let {
                                Spacer(Modifier.height(3.dp))
                                Text(it, color = Color.White.copy(alpha = 0.65f), style = T.Caption)
                            }
                        }
                        // شريط الإجراءات
                        Column(
                            Modifier.align(Alignment.BottomEnd).padding(end = 12.dp, bottom = 20.dp),
                            horizontalAlignment = Alignment.CenterHorizontally,
                            verticalArrangement = Arrangement.spacedBy(14.dp),
                        ) {
                            ReelAction(Ic.File, tr("Details", "Details", "التفاصيل")) { state.go(Screen.FileDetail(f)) }
                        }
                    }
                }
            }
        }
        ReelsTopBar(onGrid)
    }
}

@Composable
private fun ReelsTopBar(onGrid: () -> Unit) {
    Row(
        Modifier.fillMaxWidth().padding(horizontal = 14.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Text(
            tr("Reels", "Reels", "ريلز"),
            color = Color.White, style = T.Title2, fontWeight = FontWeight.SemiBold,
            modifier = Modifier.weight(1f),
        )
        IconTile(Ic.Grid, onGrid, circle = true, desc = tr("Dateien als Raster", "Files as grid", "الملفات كشبكة"))
    }
}

@Composable
private fun ReelAction(icon: androidx.compose.ui.graphics.vector.ImageVector, desc: String, onClick: () -> Unit) {
    Column(horizontalAlignment = Alignment.CenterHorizontally) {
        Box(
            Modifier.size(46.dp).clip(CircleShape).background(Color.White.copy(alpha = 0.16f)).clickable(onClick = onClick),
            contentAlignment = Alignment.Center,
        ) { Icon(icon, desc, tint = Color.White, modifier = Modifier.size(22.dp)) }
        Spacer(Modifier.height(4.dp))
        Text(desc, color = Color.White.copy(alpha = 0.85f), style = T.Micro, fontWeight = FontWeight.SemiBold)
    }
}
