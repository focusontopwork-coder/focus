/** Conversations: the client talks to the team only (never to single employees). */
@Composable
fun MessagesScreen(state: AppState) {
    var tab by remember { mutableStateOf(0) }
    var thread by remember { mutableStateOf<ChatThread?>(null) }
    var searching by remember { mutableStateOf(false) }
    var query by remember { mutableStateOf("") }
    val lang = LocalLang.current
    LaunchedEffect(Unit) { thread = runCatching { state.api.chat() }.getOrNull() }
    val home = state.home
    val last = thread?.messages?.lastOrNull()
    val q = query.trim()
    val teamPreview = last?.let { (if (it.mine) "✓ " else "") + (it.text ?: it.fileName ?: "📎") } ?: tr("Schreib uns – wir sind für dich da.", "Write to us – we're here for you.", "راسلنا، نحن هنا لخدمتك.")

    Column(Modifier.fillMaxSize().screenBg().statusBarsPadding().navigationBarsPadding()) {
        ScreenHeader(tr("Nachrichten", "Messages", "الرسائل"), onBack = { state.back() }) {
            IconTile(Ic.Pencil, { state.go(Screen.Chat()) }, gold = true)
        }
        Row(
            Modifier.padding(horizontal = 16.dp).fillMaxWidth().height(50.dp).clip(RoundedCornerShape(16.dp)).background(C.Surface.copy(alpha = 0.7f))
                .border(1.dp, C.Border, RoundedCornerShape(16.dp)).clickable { searching = true }.padding(horizontal = 14.dp),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Icon(Ic.Search, null, tint = C.Muted, modifier = Modifier.size(19.dp))
            Spacer(Modifier.width(10.dp))
            Box(Modifier.weight(1f)) {
                if (query.isEmpty()) Text(tr("Chats durchsuchen", "Search chats", "ابحث في المحادثات"), color = C.Faint, style = T.Body)
                BasicTextField(query, { query = it }, singleLine = true, textStyle = T.Body.copy(color = C.Ink), cursorBrush = SolidColor(C.Gold), modifier = Modifier.fillMaxWidth())
            }
        }
        Row(Modifier.fillMaxWidth().padding(top = 10.dp)) {
            listOf(tr("Hauptchat", "Main", "الأساسية"), tr("Team", "Team", "الفريق"), tr("Anfragen", "Requests", "الطلبات")).forEachIndexed { i, label ->
                Column(Modifier.weight(1f).clickable { tab = i }, horizontalAlignment = Alignment.CenterHorizontally) {
                    Text(label, color = if (tab == i) C.Ink else C.Muted, style = T.Body, fontWeight = if (tab == i) FontWeight.SemiBold else FontWeight.SemiBold, modifier = Modifier.padding(vertical = 10.dp))
                    Box(Modifier.fillMaxWidth().height(if (tab == i) 3.dp else 1.dp).background(if (tab == i) C.Gold else C.Border.copy(alpha = 0.5f)))
                }
            }
        }
        LazyColumn(contentPadding = PaddingValues(horizontal = 16.dp, vertical = 12.dp), verticalArrangement = Arrangement.spacedBy(6.dp)) {
            when (tab) {
                0 -> {
                    if (q.isEmpty() || "team focus on top".contains(q.lowercase()) || teamPreview.contains(q, true)) item {
                        ThreadRow({ FtBadge(56.dp, dashed = true) }, "Team Focus On Top", teamPreview, relativeTime(lang, last?.createdAt, nowEpochMs()), home?.unreadMessages ?: 0) { state.go(Screen.Chat()) }
                    }
                    home?.invoices?.firstOrNull()?.let { inv ->
                        item {
                            ThreadRow({ SystemIcon(Ic.Receipt) }, tr("Rechnungen & Buchhaltung", "Invoices & accounting", "الفواتير والمحاسبة"),
                                tr("Rechnung ${inv.invoiceNumber ?: ""} · ${fmtMoney(invoiceTotal(inv), inv.currency)}", "Invoice ${inv.invoiceNumber ?: ""} · ${fmtMoney(invoiceTotal(inv), inv.currency)}", "فاتورة ${inv.invoiceNumber ?: ""} · ${fmtMoney(invoiceTotal(inv), inv.currency)}"),
                                fmtDate(inv.issuedDate).take(6), 0) { state.go(Screen.Invoices) }
                        }
                    }
                    home?.subscription?.let { sub ->
                        item {
                            val days = home?.usage?.daysLeft
                            ThreadRow({ SystemIcon(Ic.Shield) }, tr("Abo & Tarif", "Plan & subscription", "الاشتراك والباقة"),
                                if (days != null) tr("Ihr Abo verlängert sich in $days Tagen", "Your plan renews in $days days", "يتجدد اشتراكك بعد ${arDays(days)}") else planName(state, home),
                                "", 0) { state.back(); state.tab = Tab.PROFILE }
                        }
                    }
                }
                1 -> {
                    item {
                        ThreadRow({ FtBadge(56.dp, dashed = true) }, "Team Focus On Top", tr("Alle Nachrichten gehen an das ganze Team.", "Every message reaches the whole team.", "تصل كل رسالة إلى الفريق كاملًا."), "", 0) { state.go(Screen.Chat()) }
                    }
                    home?.manager?.let { m ->
                        item {
                            ThreadRow({ InitialTile(m.name, 56.dp, m.avatar, warm = true) }, "${m.name ?: ""} · " + tr("Kundenbetreuer", "Account manager", "مدير الحساب"),
                                m.job ?: m.specialization ?: tr("Erreichbar über den Team-Chat", "Reachable via the team chat", "يمكنك التواصل معه عبر محادثة الفريق"), "", 0) { state.go(Screen.Chat()) }
                        }
                    }
                }
                else -> {
                    val reqs = home?.appointments.orEmpty().filter { it.status == "requested" }
                    if (reqs.isEmpty()) item { EmptyState(tr("Keine offenen Anfragen", "No open requests", "لا توجد طلبات مفتوحة"), icon = Ic.CalendarPlus) }
                    items(reqs, key = { it.id }) { a -> AppointmentRow(a) }
                    item { Spacer(Modifier.height(8.dp)); GoldButton(tr("Termin anfragen", "Request appointment", "طلب موعد"), { state.go(Screen.AppointmentRequest) }, Modifier.fillMaxWidth(), icon = Ic.Plus) }
                }
            }
        }
    }
}

@Composable
private fun SystemIcon(icon: ImageVector) {
    Box(Modifier.size(56.dp).clip(RoundedCornerShape(18.dp)).background(C.Surface).border(1.dp, C.Border, RoundedCornerShape(18.dp)), contentAlignment = Alignment.Center) {
        Icon(icon, null, tint = C.Gold, modifier = Modifier.size(24.dp))
    }
}

@Composable
private fun ThreadRow(avatar: @Composable () -> Unit, title: String, preview: String, time: String, unread: Int, onClick: () -> Unit) {
    Row(Modifier.fillMaxWidth().clip(RoundedCornerShape(18.dp)).clickable(onClick = onClick).padding(vertical = 10.dp), verticalAlignment = Alignment.CenterVertically) {
        avatar()
        Spacer(Modifier.width(14.dp))
        Column(Modifier.weight(1f), verticalArrangement = Arrangement.spacedBy(3.dp)) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(title, color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, maxLines = 1, overflow = TextOverflow.Ellipsis, modifier = Modifier.weight(1f))
                if (time.isNotEmpty()) Text(time, color = if (unread > 0) C.Gold else C.Muted, style = T.Micro)
            }
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(preview, color = if (unread > 0) C.Ink else C.Muted, style = T.BodyS, fontWeight = if (unread > 0) FontWeight.Bold else FontWeight.Normal, maxLines = 1, overflow = TextOverflow.Ellipsis, modifier = Modifier.weight(1f))
                if (unread > 0) {
                    Spacer(Modifier.width(6.dp))
                    Box(Modifier.size(20.dp).clip(CircleShape).background(C.Gold), contentAlignment = Alignment.Center) { Text(unread.toString(), color = C.Bg, style = T.Micro, fontWeight = FontWeight.SemiBold) }
                }
            }
        }
    }
}

/* ───────────── the team chat ───────────── */

@Composable
fun ChatScreen(state: AppState, initialDraft: String) {
    val scope = rememberCoroutineScope()
    val lang = LocalLang.current
    var messages by remember { mutableStateOf<List<ChatMessage>?>(null) }
    var draft by remember { mutableStateOf(initialDraft) }
    var busy by remember { mutableStateOf(false) }
    var uploading by remember { mutableStateOf(false) }
    val listState = rememberLazyListState()
    val failed = tr("Senden fehlgeschlagen", "Could not send", "تعذّر الإرسال")

    suspend fun load() {
        runCatching { state.api.chat() }.onSuccess { messages = it.messages; state.clearUnreadMessages() }
    }
    LaunchedEffect(Unit) { while (true) { load(); delay(8_000) } }
    LaunchedEffect(messages?.size) { messages?.let { if (it.isNotEmpty()) listState.animateScrollToItem(it.lastIndex) } }

    val picker = rememberFilePicker { files ->
        val f = files.firstOrNull() ?: return@rememberFilePicker
        uploading = true
        scope.launch {
            try {
                val blob = state.uploadBlob(f)
                state.api.sendChat(draft.trim(), blob.url, f.name, f.mimeType)
                draft = ""
                load()
            } catch (_: Exception) { state.showToast(failed) } finally { uploading = false }
        }
    }

    Column(Modifier.fillMaxSize().screenBg().statusBarsPadding().navigationBarsPadding().imePadding()) {
        Row(Modifier.fillMaxWidth().padding(horizontal = 16.dp, vertical = 10.dp), verticalAlignment = Alignment.CenterVertically) {
            IconTile(Ic.Back, { state.back() })
            Spacer(Modifier.width(10.dp))
            FtBadge(44.dp, dashed = true)
            Spacer(Modifier.width(10.dp))
            Column(Modifier.weight(1f)) {
                Text("Team Focus On Top", color = C.Ink, style = T.Headline, fontWeight = FontWeight.SemiBold, maxLines = 1)
                Text(tr("Antwortet meist innerhalb weniger Stunden", "Usually replies within a few hours", "يرد عادةً خلال ساعات قليلة"), color = C.Muted, style = T.Caption, maxLines = 1)
            }
        }
        val list = messages
        Box(Modifier.weight(1f).fillMaxWidth()) {
            when {
                list == null -> CircularProgressIndicator(Modifier.align(Alignment.Center), color = C.Gold)
                list.isEmpty() -> EmptyState(tr("Schreib dem Team", "Write to the team", "اكتب إلى الفريق"), tr("Fragen, Wünsche, Feedback – hier landet alles direkt beim Team.", "Questions, wishes, feedback – everything lands with the team.", "أسئلتك وطلباتك وملاحظاتك تصل مباشرة إلى الفريق."))
                else -> LazyColumn(state = listState, contentPadding = PaddingValues(horizontal = 16.dp, vertical = 8.dp), verticalArrangement = Arrangement.spacedBy(8.dp)) {
                    var lastDay = ""
                    list.forEach { m ->
                        val day = epochOf(m.createdAt)?.let { localDateTime(it).take(10) } ?: ""
                        if (day != lastDay && day.isNotEmpty()) {
                            lastDay = day
                            item(key = "d-$day") {
                                Box(Modifier.fillMaxWidth(), contentAlignment = Alignment.Center) {
                                    Text(dayHeader(de.focusontop.client.i18n.LocalStrings.current, day, todayIso()), color = C.Muted, style = T.Caption, modifier = Modifier.padding(vertical = 6.dp))
                                }
                            }
                        }
                        item(key = m.id) { Bubble(m, lang) }
                    }
                }
            }
        }
        Box(Modifier.padding(horizontal = 14.dp, vertical = 10.dp)) {
            ComposerBar(draft, { draft = it }, tr("Nachricht schreiben …", "Write a message …", "اكتب رسالة…"), busy || uploading, leading = {
                Box(
                    Modifier.size(52.dp).clip(RoundedCornerShape(18.dp)).background(C.Surface.copy(alpha = 0.7f)).border(1.dp, C.Border, RoundedCornerShape(18.dp))
                        .clickable(enabled = !uploading) { picker.pickFiles() },
                    contentAlignment = Alignment.Center,
                ) {
                    if (uploading) CircularProgressIndicator(Modifier.size(20.dp), color = C.Gold, strokeWidth = 2.dp)
                    else Icon(Ic.Paperclip, null, tint = C.Ink)
                }
            }) {
                val text = draft.trim()
                busy = true
                scope.launch {
                    try { state.api.sendChat(text); draft = ""; load() }
                    catch (_: ApiError) { state.showToast(failed) } finally { busy = false }
                }
            }
        }
    }
}

@Composable
private fun Bubble(m: ChatMessage, lang: de.focusontop.client.i18n.Lang) {
    val mine = m.mine
    Row(Modifier.fillMaxWidth(), horizontalArrangement = if (mine) Arrangement.End else Arrangement.Start, verticalAlignment = Alignment.Bottom) {
        if (!mine) { FtBadge(30.dp); Spacer(Modifier.width(8.dp)) }
        Column(
            Modifier.widthIn(max = 290.dp).clip(RoundedCornerShape(topStart = 20.dp, topEnd = 20.dp, bottomStart = if (mine) 20.dp else 6.dp, bottomEnd = if (mine) 6.dp else 20.dp))
                .then(if (mine) Modifier.background(GoldBrush) else Modifier.background(C.Surface).border(1.dp, C.Border.copy(alpha = 0.6f), RoundedCornerShape(topStart = 20.dp, topEnd = 20.dp, bottomStart = 6.dp, bottomEnd = 20.dp)))
                .padding(horizontal = 14.dp, vertical = 10.dp),
            verticalArrangement = Arrangement.spacedBy(4.dp),
        ) {
            if (!mine && !m.senderName.isNullOrBlank()) Text(m.senderName, color = C.Gold, style = T.Micro, fontWeight = FontWeight.SemiBold)
            m.imageUrl?.let { img ->
                AsyncImage(model = Media.image(img, 800), contentDescription = null, contentScale = ContentScale.Crop,
                    modifier = Modifier.fillMaxWidth().height(180.dp).clip(RoundedCornerShape(14.dp)).clickable { Media.abs(img)?.let(::openUrl) })
            }
            val fileUrl = m.fileData?.takeIf { it.startsWith("/uploads/") || it.startsWith("http") }
            if (fileUrl != null) {
                Row(Modifier.clip(RoundedCornerShape(12.dp)).clickable { Media.download(fileUrl, m.fileName)?.let(::openUrl) }.padding(vertical = 4.dp), verticalAlignment = Alignment.CenterVertically) {
                    Icon(Ic.Paperclip, null, tint = if (mine) C.Bg else C.Gold, modifier = Modifier.size(18.dp))
                    Spacer(Modifier.width(6.dp))
                    Text(m.fileName ?: "Datei", color = if (mine) C.Bg else C.Ink, style = T.BodyS, fontWeight = FontWeight.SemiBold, maxLines = 1, overflow = TextOverflow.Ellipsis)
                }
            }
            if (!m.text.isNullOrBlank()) Text(m.text, color = if (mine) C.Bg else C.Ink, style = T.Body)
            Text(epochOf(m.createdAt)?.let { localDateTime(it).substring(11, 16) } ?: "", color = if (mine) C.Bg.copy(alpha = 0.6f) else C.Faint, style = T.Micro)
        }
    }
}
